/* Phase 0(回归框架): 全部仿真随机数走可播种 mulberry32 —— 固定种子跑固定帧数,
 * CPU/GPU 两条路径才能做统计等价对比(Math.random 无法复现)。 */
function mulberry32(a) {
  return function () {
    a |= 0; a = a + 0x6D2B79F5 | 0;
    let t = Math.imul(a ^ a >>> 15, 1 | a);
    t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t;
    return ((t ^ t >>> 14) >>> 0) / 4294967296;
  };
}
let rand = mulberry32(0xC0FFEE);           // 仿真随机流(渲染/UI 不用它)
function setSeed(seed) { rand = mulberry32(seed >>> 0); }

// Box-Muller 高斯采样, 让初始群体成团而非均匀撒布
function randn() {
  let u = 0, v = 0;
  while (u === 0) u = rand();
  while (v === 0) v = rand();
  return Math.sqrt(-2 * Math.log(u)) * Math.cos(TAU * v);
}

/* ---- P2: 雁群的槽位几何与轮值 ------------------------------------------
 * 槽位编号 slot(i) = (i + gooseShift) % M; 槽位 0 恒为 apex(V 尖)。
 * 排号/左右臂由槽位索引决定, 但**幼雁的内道优先**靠"残类"实现:
 * 幼雁是固定个体(i%4===0, 25%), 平移量按 M=60 取模, 而 4 | 60, 所以
 * "i%4===0 的雁" 永远落在槽位残类 (gooseShift%4) 里 —— 把这一残类的槽位
 * 排成内道(小排号), 幼雁就永远在内道, 换岗也不会换出内道。 */
function rebuildFormation() {
  const juvRes = ((gooseShift % 4) + 4) % 4;
  const inner = [], outer = [];
  for (let s = 0; s < M; s++) {
    if (s === 0) continue;                  // 0 号槽固定是 apex
    if (s % 4 === juvRes) inner.push(s); else outer.push(s);
  }
  slotRank[0] = 0; slotArm[0] = 1;
  let rank = 1;
  for (let k = 0; k < inner.length; k += 2) {          // 内道: 幼雁
    slotRank[inner[k]] = rank; slotArm[inner[k]] = 1;
    if (k + 1 < inner.length) { slotRank[inner[k + 1]] = rank; slotArm[inner[k + 1]] = -1; }
    rank++;
  }
  for (let k = 0; k < outer.length; k += 2) {          // 外道: 成雁
    slotRank[outer[k]] = rank; slotArm[outer[k]] = 1;
    if (k + 1 < outer.length) { slotRank[outer[k + 1]] = rank; slotArm[outer[k + 1]] = -1; }
    rank++;
  }
}

// 雁 i 的目标槽位坐标(apex = (Lx,Ly), 航向 leaderAng)
function gooseSlot(i) {
  const s = (i + gooseShift) % M;
  const rank = slotRank[s], arm = slotArm[s];
  const back = rank * SLOT_SPACING;
  const side = arm * rank * SLOT_SPACING * 0.36;
  const hx = Math.cos(leaderAng), hy = Math.sin(leaderAng);
  return { x: wrapX(Lx - hx * back - hy * side), y: wrapY(Ly - hy * back + hx * side) };
}

// 威慑锥判定: 点 (px,py) 是否落在"以 V 尖为顶点、领队航向为轴、半角 60°、半径 180"的锥内
function inCone(px, py) {
  const dx = minImageX(px - Lx), dy = minImageY(py - Ly);
  const d2 = dx * dx + dy * dy;
  if (d2 > CONE_R * CONE_R) return false;
  const d = Math.sqrt(d2) || 1;
  return (dx / d) * Math.cos(leaderAng) + (dy / d) * Math.sin(leaderAng) > CONE_COS;
}

// 雁 i 的死亡流程: 幼雁 -> 全体沮丧期(P2C); apex -> 斩首混乱期(P2E)。
// 被吃与调试钩子(SIM.killGoose)共用同一条路径, 保证测试测的就是真实行为。
function gooseDies(i) {
  if (!galive[i]) return;
  addDeath(P2[2*i], P2[2*i+1]);
  galive[i] = 0; galiveCount--;
  if (isJuvenile(i)) { gooseDebuff = JUVENILE_DEBUFF_FRAMES; stats.juvenileLost++; }
  if (i === leaderIdx) {
    disarray = DISARRAY_FRAMES;
    stats.gooseDisarray++;
    rotateLeader();                            // 先认一个新 apex, 混乱结束后按它重建
  }
}

// 把 apex 换到下一只活雁(轮值 / 混乱后重建); 返回是否成功。
// 不变式: slot(leaderIdx) = (leaderIdx + gooseShift) % M = 0。
// 要把 apex 前进 k 只, 平移量必须**减** k(因为 leaderIdx = -gooseShift mod M)。
function rotateLeader() {
  for (let k = 1; k <= M; k++) {
    const cand = (leaderIdx + k) % M;
    if (galive[cand]) {
      gooseShift = ((gooseShift - k) % M + M) % M;   // 槽位序号平移
      leaderIdx = cand;
      rebuildFormation();
      return true;
    }
  }
  return false;
}

function initGeese() {
  galive.fill(1); galiveCount = M;
  leaderAng = rand() * TAU;
  Lx = wrapX(W * 0.3); Ly = wrapY(H * 0.3);
  gooseShift = 0; leaderIdx = 0; disarray = 0; gooseDebuff = 0;
  juvCount = 0;
  for (let i = 0; i < M; i++) if (isJuvenile(i)) juvCount++;
  rebuildFormation();
  const hx = Math.cos(leaderAng), hy = Math.sin(leaderAng);
  for (let i = 0; i < M; i++) {
    const s = gooseSlot(i);
    P2[2*i] = s.x; P2[2*i+1] = s.y;
    V2[2*i] = hx * LEADER_SPEED; V2[2*i+1] = hy * LEADER_SPEED;
  }
  Lx = P2[2*leaderIdx]; Ly = P2[2*leaderIdx+1];   // 虚拟领队 = apex 雁当前位置
}

function resetFlocks() { initStarlings(); initGeese(); }

  f1.x = W / 2; f1.y = inOpen ? H + 40 : H * 0.85; f1.vx = 0; f1.vy = -CIRCLE_SPEED; f1.circleT = 200;

/* --------------------- 4. 自动补充猎物 --------------------- */

// 四个补充方向(常量, 避免每次补充时新建对象字面量数组)
const REFILL_SIDES = [
  { x: -PAD - 20, y: H / 2, inx: 1, iny: 0 },
  { x: W + PAD + 20, y: H / 2, inx: -1, iny: 0 },
  { x: W / 2, y: -PAD - 20, inx: 0, iny: 1 },
  { x: W / 2, y: H + PAD + 20, inx: 0, iny: -1 },
];

// 死亡太多时, 从离猛禽最远的一侧补一波: 让竞赛能长时间看下去。
// 两趟扫描(先挑边, 再就地安置), 不再收集 dead 索引数组。
function refillStarlings() {
  let deadCount = 0;
  for (let i = 0; i < N; i++) if (!alive[i]) deadCount++;
  if (deadCount === 0) return;

  let best = REFILL_SIDES[0], bestD = -1;
  for (let s = 0; s < REFILL_SIDES.length; s++) {
    const sd = REFILL_SIDES[s];
    let dmin = 1e18;
    for (const f of threats) {
      const dx = minImageX(sd.x - f.x), dy = minImageY(sd.y - f.y);
      dmin = Math.min(dmin, dx * dx + dy * dy);
    }
    if (dmin > bestD) { bestD = dmin; best = sd; }
  }

  for (let i = 0; i < N; i++) {
    if (alive[i]) continue;
    const jx = best.inx !== 0 ? randn() * 25 : randn() * 200;
    const jy = best.iny !== 0 ? randn() * 25 : randn() * 200;
    P[2*i]   = wrapX(best.x + jx);
    P[2*i+1] = wrapY(best.y + jy);
    const vx = best.inx * MAX_SPEED * 0.6 + (rand() - 0.5) * 0.6;
    const vy = best.iny * MAX_SPEED * 0.6 + (rand() - 0.5) * 0.6;
    const vs = Math.hypot(vx, vy) || 1;
    V[2*i]   = vx / vs * MAX_SPEED * 0.6;
    V[2*i+1] = vy / vs * MAX_SPEED * 0.6;
    PV[2*i] = V[2*i]; PV[2*i+1] = V[2*i+1];
    startled[i] = 0;
    alive[i] = 1;
  }
  aliveCount += deadCount;
  stats.wave++;
  gpuSyncBirds();                            // GPU 路径: 复活后的 alive/位置全量上行
}

function refillGeese() {
  const hx = Math.cos(leaderAng), hy = Math.sin(leaderAng);
  let added = 0;
  for (let i = 0; i < M; i++) {
    if (galive[i]) continue;
    const s = gooseSlot(i);
    P2[2*i] = s.x; P2[2*i+1] = s.y;
    V2[2*i] = hx * LEADER_SPEED; V2[2*i+1] = hy * LEADER_SPEED;
    galive[i] = 1; added++;
  }
  galiveCount += added;
}

function maybeRefill() {
  if (!refillOn) return;
  if (aliveCount < N * REFILL_RATIO) refillStarlings();
  if (galiveCount < M * REFILL_G_RATIO) refillGeese();
}

/* --------------------- 5. 猛禽感知与采样目标获取(无全局视野) --------------------- */

// 采样目标获取(P1)用的预计算量与复用缓冲
const CANDX = new Float64Array(N_CAND), CANDY = new Float64Array(N_CAND);
const CAND_R = new Float64Array(N_CAND);
for (let r = 0, k = 0; r < SAMPLE_RADII.length; r++) {
  for (let a = 0; a < SAMPLE_ANGLES; a++, k++) {
    const ang = a / SAMPLE_ANGLES * TAU;
    CANDX[k] = Math.cos(ang) * SAMPLE_RADII[r];
    CANDY[k] = Math.sin(ang) * SAMPLE_RADII[r];
    CAND_R[k] = SAMPLE_RADII[r];
  }
}
const candScore = new Float64Array(N_CAND);
const VIS = new Int32Array(MAXN);          // 视野内可见椋鸟索引(复用)
const ANG_BIN = TAU / SAMPLE_ANGLES;       // 每个候选角度占的弧度
// 每个环的"角度窗"半宽(单位: 角度槽数), 取 bd=ring 时的最坏情况:
//   Δθ_max = 2·asin(SCOOP_R / (2·ring)), 槽数 = ceil(Δθ_max / ANG_BIN)
// 这样内层循环里不用再做 asin/sqrt, 只需 1 次 atan2(每只鸟)+ 若干平方比较。
const CAND_W = new Int32Array(SAMPLE_RADII.length);
const CAND_LO2 = new Float64Array(SAMPLE_RADII.length), CAND_HI2 = new Float64Array(SAMPLE_RADII.length);
for (let r = 0; r < SAMPLE_RADII.length; r++) {
  const rad = SAMPLE_RADII[r];
  const half = 2 * Math.asin(Math.min(1, SCOOP_R / (2 * rad)));
  CAND_W[r] = Math.min(Math.max(1, Math.ceil(half / ANG_BIN)), SAMPLE_ANGLES >> 1);
  const lo = Math.max(0, rad - SCOOP_R), hi = rad + SCOOP_R;
  CAND_LO2[r] = lo * lo; CAND_HI2[r] = hi * hi;
}

// perceive: 视野内的椋鸟统计 + 采样目标获取。
//   1) 沿用桶心粗筛 + 逐只精确判距, 收集全部可见椋鸟(质心 x,y / 数量 n 照旧);
//   2) 每只鸟累加进"距离 < SCOOP_R 的所有候选点"(80 个极坐标候选), 得到
//      "哪个位置一小勺能捞到最多鸟";
//   3) 打分 = 捞到的鸟数 ×(1 + DIR_LAG·cos(候选方向, 当前航向)) —— 方向滞后防抖动;
//      // 扩展位: 雁权重 / 威慑锥扣分 / mob 扣分 都可以在这里乘上 (P2D 已接锥)
//   4) argmax 候选点 = 冲刺中心 (cx,cy); lockIdx = 中心 LOCK_R 内最近的鸟,
//      40px 内没鸟则视为"目标不可用"(caller 会放弃这次冲刺)。
// bi 语义改为"中心附近那只代表鸟"(= lockIdx), 供伏击按它的速度外推。
// 复用的返回对象(每帧被猛禽调用 3 次 + 冲刺收尾 1 次): 不再每次分配对象字面量
const PC = { x: 0, y: 0, n: 0, bi: -1, cx: -1, cy: -1, lockIdx: -1, score: 0 };

function perceive(px, py, vx, vy, viewR2) {
  const cellR = Math.sqrt(viewR2) + CELL * 0.75, cellR2 = cellR * cellR;
  const maxCandR = Math.sqrt(viewR2) - SCOOP_R;     // 捞勺伸出视野的候选点直接不要
  let cnt = 0, nVis = 0;
  let sumX = 0, sumY = 0, scx = 0, ssx = 0, scy = 0, ssy = 0;
  candScore.fill(0);

  // 按"桶首次出现顺序"遍历(位级等价于旧版的 Map.forEach), 桶内逆链成升序
  for (let oi = 0; oi < cellOrderN; oi++) {
    const c = cellOrder[oi];
    const cx = c % GW, cy = (c / GW) | 0;
    const bx = -PAD + (cx + 0.5) * CELL, by = -PAD + (cy + 0.5) * CELL;
    const dx = minImageX(bx - px), dy = minImageY(by - py);
    if (dx * dx + dy * dy > cellR2) continue;
    let m = 0;
    for (let j = gridHead[c]; j !== -1; j = gridNext[j]) chainBuf[m++] = j;   // 降序收集
    for (let q = m - 1; q >= 0; q--) {                                        // 升序遍历
      const j = chainBuf[q];
      const ex = minImageX(P[2*j] - px), ey = minImageY(P[2*j+1] - py);
      const d2 = ex * ex + ey * ey;
      if (d2 > viewR2) continue;
      cnt++;
      if (nVis < MAXN) VIS[nVis++] = j;
      if (TORUS) {
        const ta = P[2*j] / W * TAU, tb = P[2*j+1] / H * TAU;
        scx += Math.cos(ta); ssx += Math.sin(ta);
        scy += Math.cos(tb); ssy += Math.sin(tb);
      } else { sumX += P[2*j]; sumY += P[2*j+1]; }

      // 这只鸟能被哪些候选点"捞"到? 两层预筛把 80 个候选砍到 5~10 个:
      //   1) 半径: |ring - bd| < SCOOP_R(用平方比较, 不 sqrt);
      //   2) 角度: 只试最近角度槽 ±CAND_W[ring] 个(最坏情况窗, 预计算)。
      const bang = Math.atan2(ey, ex);
      const c0 = Math.round(bang / ANG_BIN);
      for (let r = 0, k = 0; r < SAMPLE_RADII.length; r++, k += SAMPLE_ANGLES) {
        if (SAMPLE_RADII[r] > maxCandR || d2 < CAND_LO2[r] || d2 > CAND_HI2[r]) continue;
        const w = CAND_W[r];
        for (let o = -w; o <= w; o++) {
          let ai = (c0 + o) % SAMPLE_ANGLES; if (ai < 0) ai += SAMPLE_ANGLES;
          const kk = k + ai;
          const ddx = ex - CANDX[kk], ddy = ey - CANDY[kk];
          if (ddx * ddx + ddy * ddy < SCOOP_R2) candScore[kk] += 1;
        }
      }
    }
  }

  if (cnt === 0) return null;
  const mx = TORUS ? wrapX(Math.atan2(ssx, scx) / TAU * W) : sumX / cnt;
  const my = TORUS ? wrapY(Math.atan2(ssy, scy) / TAU * H) : sumY / cnt;

  // 打分: 方向滞后 + 威慑锥乘项(P2D: 让选目标自己绕开雁群正面, 不写躲避硬编码)
  const sp = Math.sqrt(vx * vx + vy * vy) || 1;
  const ux = vx / sp, uy = vy / sp;
  const coneOn = galiveCount > 0;
  let bestS = 0, bestK = -1;
  for (let k = 0; k < N_CAND; k++) {
    const base = candScore[k];
    if (base <= 0 || CAND_R[k] > maxCandR) continue;
    const d = CAND_R[k] || 1;
    let sc = base * (1 + DIR_LAG * ((CANDX[k] / d) * ux + (CANDY[k] / d) * uy));
    // 扩展位: 雁权重 / 威慑锥扣分 / mob 扣分
    if (coneOn && inCone(px + CANDX[k], py + CANDY[k])) sc *= CONE_SCORE_FACTOR;
    if (sc > bestS) { bestS = sc; bestK = k; }
  }

  let cxT = -1, cyT = -1, lockIdx = -1;
  if (bestK >= 0) {
    cxT = px + CANDX[bestK]; cyT = py + CANDY[bestK];
    let bestD = LOCK_R2;                     // 中心附近最近的鸟: 冲刺途中只吃它
    for (let v = 0; v < nVis; v++) {
      const j = VIS[v];
      const dx = minImageX(P[2*j] - cxT), dy = minImageY(P[2*j+1] - cyT);
      const d2 = dx * dx + dy * dy;
      if (d2 < bestD) { bestD = d2; lockIdx = j; }
    }
    if (lockIdx < 0) { cxT = -1; cyT = -1; } // 40px 内无鸟 -> 本次放弃
  }
  PC.x = mx; PC.y = my; PC.n = cnt; PC.bi = lockIdx;
  PC.cx = cxT; PC.cy = cyT; PC.lockIdx = lockIdx; PC.score = bestS;
  return PC;
}

// tryDash: 发起冲刺。带 ±60° 方向约束 —— 目标必须在当前航向 60° 以内,
// 否则放弃并返回 false(盘旋转向会逐渐对齐, 这是特性不是 bug)。
// lockIdx: D 的迷惑效应 —— 冲刺锁定的椋鸟索引(雁/对手传 -1, 不算锁定)。
function tryDash(f, tx, ty, lockIdx) {
  const dx = minImageX(tx - f.x), dy = minImageY(ty - f.y);
  const n = Math.hypot(dx, dy);
  if (n <= 1) return false;
  const curA = Math.atan2(f.vy, f.vx), tgtA = Math.atan2(dy, dx);
  if (Math.abs(angDiff(tgtA, curA)) > DASH_MAX_ANGLE) return false;
  f.vx = dx / n * DASH_SPEED;
  f.vy = dy / n * DASH_SPEED;
  f.dashT = Math.floor((Math.min(n, DASH_MAX_DIST) + DASH_EXTRA) / DASH_SPEED);
  f.shock = { x: tx, y: ty, t: 0 };              // 冲刺瞬间的冲击波
  f.dashEats = 0;                                // A: 本次冲刺的猎物计数清零
  f.dashS = 0;                                   // P1: 本次冲刺吃到的椋鸟数(验收用)
  f.lockIdx = (DASH_LOCK && lockIdx !== undefined) ? lockIdx : -1;
  if (f.lockIdx >= 0) stats.lockedDashes++;
  return true;
}

/* ------------------------------ 6. 仿真 step ------------------------------ */

let migAng = rand() * TAU;          // 群体迁徙方向(缓慢随机游动)

/* ---- 质心/迁徙的模块级状态 + 6.1/6.3 提出为函数(CPU 全路径与 GPU 路径共用) ---- */
let cxm = W / 2, cym = H / 2;              // 全局质心(6.2 或 GPU 回读写入)
let migX = 0, migY = 0;                    // 迁徙力(doMigrate 写入; 6.4 与 GPU uniform 读取)

/* 6.1 网格重建(提出为函数: GPU 路径里 perceive 每帧仍要用网格预筛, 仅 ~5% 留在 CPU) */
function rebuildGrid() {
  /* 链表是头插法(桶内降序), 但**遍历顺序必须与旧版逐位一致**, 否则浮点求和
   * 顺序变了、混沌系统会放大成完全不同的轨迹: 所以额外记住"桶的首次出现顺序"
   * (cellOrder)并在遍历时把链逆成升序(chainBuf) —— 位级复现旧 Map+数组的行为。 */
  gridHead.fill(-1);
  cellOrderN = 0;
  for (let i = 0; i < N; i++) {
    if (!alive[i]) continue;
    const c = cellOf(P[2*i], P[2*i+1]);
    if (gridHead[c] === -1) cellOrder[cellOrderN++] = c;
    gridNext[i] = gridHead[c];               // 头插法: i 成为该桶新的链头
    gridHead[c] = i;
  }
  for (const f of threats) f.mob = 0;        // C: 围攻计数每帧清零, 本帧重新统计
}

/* 6.3 迁徙方向: 随机游动 + 离屏太远就掰回来(GPU 路径的质心来自上一帧摘要, 一帧延迟无感) */
function doMigrate() {
  migAng += (rand() - 0.5) * 0.025;
  const offX = TORUS ? minImageX(W / 2 - cxm) : W / 2 - cxm;
  const offY = TORUS ? minImageY(H / 2 - cym) : H / 2 - cym;
  if (Math.sqrt(offX * offX + offY * offY) > 150) {
    migAng += angDiff(Math.atan2(offY, offX), migAng) * 0.02;
  }
  migX = Math.cos(migAng) * MIGRATE_FORCE;
  migY = Math.sin(migAng) * MIGRATE_FORCE;
}

function step() {
  frame++; stats.steps++;
  const tor = TORUS;                         // 每步取一次, 热循环里不再反复查全局
  rebuildGrid();
  /* ---- 6.2 椋鸟: 拓扑邻居扫描(只算力) + 全局质心累加 ---- */
  let sumX = 0, sumY = 0, cntAlive = 0;
  let sXc = 0, cXc = 0, sYc = 0, cYc = 0;

  for (let i = 0; i < N; i++) {
    if (!alive[i]) continue;
    const xi = P[2*i], yi = P[2*i+1];
    const cx = clamp(Math.floor((xi + PAD) / CELL), 0, GW - 1);
    const cy = clamp(Math.floor((yi + PAD) / CELL), 0, GH - 1);

    // 收集候选, 只保留最近的 topoK 只(拓扑邻居; A 版的核心)。
    // 槽位按距离升序维护: 比"每次重扫最远槽"少一个 O(K) 内层循环,
    // 大多数候选只需一次比较 KD[kn-1] 就被丢掉。
    let kn = 0;
    for (let ox = -1; ox <= 1; ox++)
      for (let oy = -1; oy <= 1; oy++) {
        let gx = cx + ox, gy = cy + oy;
        if (tor) { gx = ((gx % GW) + GW) % GW; gy = ((gy % GH) + GH) % GH; }
        else if (gx < 0 || gx >= GW || gy < 0 || gy >= GH) continue;
        for (let j = gridHead[gy * GW + gx]; j !== -1; j = gridNext[j]) {
          if (j === i) continue;
          const dx = tor ? minImageX(P[2*j] - xi) : P[2*j] - xi;
          const dy = tor ? minImageY(P[2*j+1] - yi) : P[2*j+1] - yi;
          const d2 = dx * dx + dy * dy;
          if (d2 > PERCEPTION_R2) continue;
          let p;
          if (kn < topoK) p = kn++;            // 先填满 K 个槽
          else if (d2 < KD[kn-1]) p = kn - 1;  // 比当前最远的还近 -> 顶掉它
          else continue;
          while (p > 0 && KD[p-1] > d2) {      // 冒泡到正确位置, 维持升序
            KD[p] = KD[p-1]; KX[p] = KX[p-1]; KY[p] = KY[p-1]; KJ[p] = KJ[p-1];
            p--;
          }
          KD[p] = d2; KX[p] = dx; KY[p] = dy; KJ[p] = j;
        }
      }

    let ax = 0, ay = 0;
    if (kn > 0) {
      let psx = 0, psy = 0, vsx = 0, vsy = 0, sepx = 0, sepy = 0;
      for (let m = 0; m < kn; m++) {
        psx += KX[m]; psy += KY[m];
        vsx += V[2*KJ[m]]; vsy += V[2*KJ[m]+1];
        if (KD[m] < SEP_R2) {                  // 距离加权分离: 越近推得越狠
          const w = 1 / Math.max(KD[m], 1);
          sepx -= KX[m] * w; sepy -= KY[m] * w;
        }
      }
      const mx = COHESION_W * psx / kn + ALIGNMENT_W * vsx / kn;
      const my = COHESION_W * psy / kn + ALIGNMENT_W * vsy / kn;
      const nn = Math.hypot(mx, my);
      if (nn > 1e-9) {
        ax += clamp(mx / nn * MAX_SPEED - V[2*i],   -MAX_FORCE, MAX_FORCE);
        ay += clamp(my / nn * MAX_SPEED - V[2*i+1], -MAX_FORCE, MAX_FORCE);
      }
      ax += clamp(sepx * SEP_GAIN, -2 * MAX_FORCE, 2 * MAX_FORCE);
      ay += clamp(sepy * SEP_GAIN, -2 * MAX_FORCE, 2 * MAX_FORCE);
    }
    ACC[2*i] = ax; ACC[2*i+1] = ay;

    sumX += xi; sumY += yi; cntAlive++;
    if (tor) {                                 // 环面质心走圆均值
      sXc += Math.sin(xi / W * TAU); cXc += Math.cos(xi / W * TAU);
      sYc += Math.sin(yi / H * TAU); cYc += Math.cos(yi / H * TAU);
    }
  }

  cxm = cntAlive === 0 ? W / 2
      : (tor ? wrapX(Math.atan2(sXc, cXc) / TAU * W) : sumX / cntAlive);
  cym = cntAlive === 0 ? H / 2
      : (tor ? wrapY(Math.atan2(sYc, cYc) / TAU * H) : sumY / cntAlive);

  doMigrate();
  /* ---- 6.4 椋鸟: 积分(质心引力 + 迁徙 + 边界 + 被捕食 + 逃逸) ---- */
  for (let i = 0; i < N; i++) {
    if (!alive[i]) continue;
    const xi = P[2*i], yi = P[2*i+1];
    const vi = 2 * i;
    let ax = ACC[vi], ay = ACC[vi+1];

    const dcx = tor ? minImageX(cxm - xi) : cxm - xi;
    const dcy = tor ? minImageY(cym - yi) : cym - yi;
    const dn = Math.sqrt(dcx * dcx + dcy * dcy);   // 弱质心引力, 只防永久碎裂
    if (dn > 1e-9) { ax += dcx / dn * CENTROID_FORCE; ay += dcy / dn * CENTROID_FORCE; }

    ax += migX; ay += migY;

    if (!tor) {                                // 开阔世界的柔性边界
      if (xi < -PAD) ax += BOUND_FORCE;
      else if (xi > W + PAD) ax -= BOUND_FORCE;
      if (yi < -PAD) ay += BOUND_FORCE;
      else if (yi > H + PAD) ay -= BOUND_FORCE;
    }

    let eaten = false;                         // 被捕食: 撞上任意猛禽即消失
    for (let t = 0; t < threats.length; t++) {
      const f = threats[t];
      // D 迷惑效应: 冲刺锁定某只鸟时, 冲刺期间只吃得到被锁定的那一只
      if (f.dashT > 0 && f.lockIdx >= 0 && f.lockIdx !== i) continue;
      const ex = tor ? minImageX(xi - f.x) : xi - f.x;
      const ey = tor ? minImageY(yi - f.y) : yi - f.y;
      if (ex * ex + ey * ey < EAT_R2_S) {
        f.score += SCORE_STARLING; f.pending += SCORE_STARLING; f.kills++;
        if (f.dashT > 0) { f.dashEats++; f.dashS++; }   // A: 吃到东西 / P1 验收只数椋鸟
        if (f === eagle) eagle.dietStarling++;
        stats.kills++;
        alive[i] = 0; aliveCount--;
        addDeath(xi, yi);
        eaten = true;
        break;
      }
    }
    if (eaten) continue;

    for (let t = 0; t < threats.length; t++) {  // 逃逸: 视野内所有猛禽都生效
      const f = threats[t];
      const dpx = tor ? minImageX(xi - f.x) : xi - f.x;
      const dpy = tor ? minImageY(yi - f.y) : yi - f.y;
      const dp2 = dpx * dpx + dpy * dpy;
      if (dp2 < FLEE_R2 && dp2 > 0) {
        const dp = Math.sqrt(dp2);
        ax += FLEE_W * clamp(dpx / dp * MAX_SPEED - V[vi],   -FLEE_FORCE, FLEE_FORCE);
        ay += FLEE_W * clamp(dpy / dp * MAX_SPEED - V[vi+1], -FLEE_FORCE, FLEE_FORCE);
      }
    }

    // C 围攻: 围攻者逼近猛禽到骚扰环 —— 环外往里压, 环内往外推, 鸟群围着猛禽转
    if (isMobber[i]) {
      let mf = null, md2 = 1e18, mdx = 0, mdy = 0;
      for (let t = 0; t < threats.length; t++) {
        const f = threats[t];
        const dx = tor ? minImageX(xi - f.x) : xi - f.x;
        const dy = tor ? minImageY(yi - f.y) : yi - f.y;
        const d2 = dx * dx + dy * dy;
        if (d2 < md2) { md2 = d2; mf = f; mdx = dx; mdy = dy; }
      }
      const md = Math.sqrt(md2);
      if (mf && md < MOB_R && md > 1e-6) {
        mf.mob++;                              // 本帧被这只鸟围住
        const ux = mdx / md, uy = mdy / md;    // u = "猛禽指向鸟"的单位向量
        const push = (md - MOB_HARASS) / MOB_HARASS;
        const tx2 = -ux * push * MAX_SPEED * 1.5;
        const ty2 = -uy * push * MAX_SPEED * 1.5;
        ax += clamp(tx2 - V[vi],   -MOB_MAX, MOB_MAX);
        ay += clamp(ty2 - V[vi+1], -MOB_MAX, MOB_MAX);
      }
    }

    let vx = V[vi] + ax, vy = V[vi+1] + ay;
    const sp = Math.sqrt(vx * vx + vy * vy);
    if (sp > MAX_SPEED) { vx = vx / sp * MAX_SPEED; vy = vy / sp * MAX_SPEED; }
    V[vi] = vx; V[vi+1] = vy;
    P[vi]   = wrapX(xi + vx);
    P[vi+1] = wrapY(yi + vy);

    // P4 惊飞判据: **必须放在这里** —— 位置/速度已经更新, 还没覆盖 PV。
    // 判据是"相邻两帧速度夹角 > 60°"(急转, 不是快慢: 椋鸟平时也贴满速)。
    // 顺序错一帧(先覆盖 PV 再比)会让全体互相拿自己比, 全员误报。
    const pvx = PV[vi], pvy = PV[vi+1];
    const np = Math.sqrt(pvx * pvx + pvy * pvy);
    if (np > 1e-6 && sp > 1e-6) {
      startled[i] = ((vx * pvx + vy * pvy) / (sp * np) < PV_ALERT_COS) ? 1 : 0;
    } else startled[i] = 0;
    PV[vi] = vx; PV[vi+1] = vy;
  }

  stepTail();
}

/* ---- 6.5-6.10: 雁群 / 鹰 Q / 隼 ES 与轮换 / 补充 / 互攻(全部留 CPU, 与计算后端无关) ---- */
function stepTail() {
  /* ---- 6.5 雁群: 中心化秩序(领队智能 / 轮值 / 幼雁内道 / 威慑锥 / 斩首混乱)
   *   + 第四部分: 借椋鸟群的惊飞网提前预警 ---- */

  // --- 轮值领队: 每 ROTATE_FRAMES 换一次 apex; 虚拟领队位置 = apex 雁当前位置(不瞬移)
  if (frame % ROTATE_FRAMES === 0) rotateLeader();
  if (disarray > 0) disarray--;
  if (gooseDebuff > 0) gooseDebuff--;
  Lx = P2[2*leaderIdx]; Ly = P2[2*leaderIdx+1];       // apex 自己就是 V 尖

  // --- 领队智能(A): 感知 400px 内最近猛禽 -> 限速转向远离; 无威胁时随机游动 + 向心
  let leaderHustle = 1;
  {
    let ld = 1e18, lx = 0, ly = 0;
    for (let t = 0; t < threats.length; t++) {
      const f = threats[t];
      const dx = minImageX(Lx - f.x), dy = minImageY(Ly - f.y);
      const d2 = dx * dx + dy * dy;
      if (d2 < ld) { ld = d2; lx = dx; ly = dy; }
    }
    const ld2 = Math.sqrt(ld);
    if (ld2 < LEADER_SENSE_R && ld2 > 1e-6) {
      const away = Math.atan2(ly, lx);                 // 远离猛禽的方向
      leaderAng += clamp(angDiff(away, leaderAng), -LEADER_MAX_TURN, LEADER_MAX_TURN);
    } else {
      leaderAng += (rand() - 0.5) * 0.02;       // 随机游动
      const ox = minImageX(W / 2 - Lx), oy = minImageY(H / 2 - Ly);
      if (Math.hypot(ox, oy) > 300) leaderAng += angDiff(Math.atan2(oy, ox), leaderAng) * 0.015;
    }
    // 威胁很近了: 领队带头加速(队形不变, 只是整体提速)
    if (ld2 < LEADER_PANIC_R) leaderHustle = ESCAPE_HUSTLE;
  }
  const lhx = Math.cos(leaderAng), lhy = Math.sin(leaderAng);
  const slotScale = (gooseDebuff > 0 ? DEBUFF_FACTOR : 1);

  // P2A 验收取样(每 30 帧): 雁尖与最近隼的距离 —— 领队智能有没有把队形带开
  if (frame % 30 === 0 && galiveCount > 0) {
    let bd = 1e18;
    for (const f of falcons) {
      const dx = minImageX(Lx - f.x), dy = minImageY(Ly - f.y);
      const d2 = dx * dx + dy * dy;
      if (d2 < bd) bd = d2;
    }
    bd = Math.sqrt(bd);
    stats.apexDistSum += bd; stats.apexDistSamples++;
    if (bd < 300) stats.apexClose++;
  }

  for (let i = 0; i < M; i++) {
    if (!galive[i]) continue;
    const xi = P2[2*i], yi = P2[2*i+1];

    let nd = 1e18, nx = 0, ny = 0;             // 最近的猛禽(含鹰)
    for (const f of threats) {
      const dx = minImageX(xi - f.x), dy = minImageY(yi - f.y);
      const d2 = dx * dx + dy * dy;
      if (d2 < nd) { nd = d2; nx = dx; ny = dy; }
    }
    const d = Math.sqrt(nd);
    const raptorNear = d < FORMATION_BREAK && d > 0;

    // --- P4 借网预警: 查周边 3×3 桶里 140px 内的惊飞椋鸟, >=3 只视同猛禽逼近
    let alert = false, alx = 0, aly = 0, aln = 0;
    {
      const cx = clamp(Math.floor((xi + PAD) / CELL), 0, GW - 1);
      const cy = clamp(Math.floor((yi + PAD) / CELL), 0, GH - 1);
        for (let ox = -1; ox <= 1; ox++)
        for (let oy = -1; oy <= 1; oy++) {
          let gx = cx + ox, gy = cy + oy;
          if (TORUS) { gx = ((gx % GW) + GW) % GW; gy = ((gy % GH) + GH) % GH; }
          else if (gx < 0 || gx >= GW || gy < 0 || gy >= GH) continue;
          const c2 = gy * GW + gx;
          let m = 0;                                          // 逆链成升序(与旧版同序)
          for (let j = gridHead[c2]; j !== -1; j = gridNext[j]) chainBuf[m++] = j;
          for (let q = m - 1; q >= 0; q--) {
            const j = chainBuf[q];
            if (!startled[j]) continue;
            const dx = minImageX(P[2*j] - xi), dy = minImageY(P[2*j+1] - yi);
            if (dx * dx + dy * dy > ALERT_R2) continue;
            aln++; alx += V[2*j]; aly += V[2*j+1];      // 跟流: 惊飞鸟的平均速度方向
          }
        }
      if (aln >= ALERT_N) {
        alert = true;
        stats.gooseAlerts++;
        // 附近 400px 内没有猛禽却报警 = 误报(涌现代价: 椋鸟翻滚也会触发)
        if (d > ALERT_FALSE_R) stats.gooseFalseAlerts++;
      }
    }

    let ax = 0, ay = 0;
    if (disarray > 0) {
      // E 斩首混乱期: 槽位力关闭, 全体退化为个体径向逃逸(复用逃逸路径)
      if (d > 1e-6) {
        ax += FLEE_W * 1.2 * clamp(nx / d * MAX_SPEED_G - V2[2*i],   -FLEE_FORCE, FLEE_FORCE);
        ay += FLEE_W * 1.2 * clamp(ny / d * MAX_SPEED_G - V2[2*i+1], -FLEE_FORCE, FLEE_FORCE);
      }
    } else if (raptorNear) {
      // 编队让位: 逃逸优先, V 形暂时解散(雁的"配合"在此刻让位于保命)
      ax += FLEE_W * 1.2 * clamp(nx / d * MAX_SPEED_G - V2[2*i],   -FLEE_FORCE, FLEE_FORCE);
      ay += FLEE_W * 1.2 * clamp(ny / d * MAX_SPEED_G - V2[2*i+1], -FLEE_FORCE, FLEE_FORCE);
    } else if (alert) {
      // 借网预警: 不靠自己的眼睛, 直接跟着惊飞椋鸟的流向走(它们逃向 = 远离猛禽)
      const an = Math.hypot(alx, aly) || 1;
      const tx2 = alx / an * MAX_SPEED_G, ty2 = aly / an * MAX_SPEED_G;
      ax += FLEE_W * 1.2 * clamp(tx2 - V2[2*i],   -FLEE_FORCE, FLEE_FORCE);
      ay += FLEE_W * 1.2 * clamp(ty2 - V2[2*i+1], -FLEE_FORCE, FLEE_FORCE);
    } else {
      // 归位: 朝自己的槽位飞(槽位由轮值平移 + 幼雁内道决定)。
      // 内联槽位计算(原 gooseSlot(i) 每帧给 60 只雁各分配一个 {x,y} 对象);
      // 注意必须与旧版逐位一致: wrapX/wrapY 先收拢槽位坐标, 再算最小像差。
      const sl = (i + gooseShift) % M;
      const rk = slotRank[sl], side = slotArm[sl] * rk * SLOT_SPACING * 0.36;
      const back = rk * SLOT_SPACING;
      const sxT = wrapX(Lx - lhx * back - lhy * side);
      const syT = wrapY(Ly - lhy * back + lhx * side);
      const gx2 = minImageX(sxT - xi), gy2 = minImageY(syT - yi);
      const gd = Math.hypot(gx2, gy2);
      if (gd > 4) {
        ax += slotScale * clamp(gx2 / gd * LEADER_SPEED - V2[2*i],   -SLOT_FORCE, SLOT_FORCE);
        ay += slotScale * clamp(gy2 / gd * LEADER_SPEED - V2[2*i+1], -SLOT_FORCE, SLOT_FORCE);
      }
      ax += 0.5 * clamp(lhx * LEADER_SPEED - V2[2*i],   -MAX_FORCE, MAX_FORCE);
      ay += 0.5 * clamp(lhy * LEADER_SPEED - V2[2*i+1], -MAX_FORCE, MAX_FORCE);
    }

    for (let j = 0; j < M; j++) {              // 雁间弱分离(O(M²), M 小无所谓)
      if (j === i || !galive[j]) continue;
      const dx = minImageX(xi - P2[2*j]), dy = minImageY(yi - P2[2*j+1]);
      const d2 = dx * dx + dy * dy;
      if (d2 < GSEP_R2 && d2 > 0) {
        const dd = Math.sqrt(d2);
        ax += dx / dd * 0.05; ay += dy / dd * 0.05;
      }
    }

    if (!TORUS) {
      if (xi < -PAD) ax += BOUND_FORCE;
      else if (xi > W + PAD) ax -= BOUND_FORCE;
      if (yi < -PAD) ay += BOUND_FORCE;
      else if (yi > H + PAD) ay -= BOUND_FORCE;
    }

    let eaten = false;                         // 猛禽顺路捕食: 雁值 100 分
    for (const f of threats) {
      const ex = minImageX(xi - f.x), ey = minImageY(yi - f.y);
      if (ex * ex + ey * ey < EAT_R2_G) {
        f.score += SCORE_GOOSE; f.pending += SCORE_GOOSE; f.kills++;
        if (f.dashT > 0) f.dashEats++;         // A: 这次冲刺确实吃到了东西(雁也算)
        if (f === eagle) eagle.dietGoose++;
        stats.kills++;
        gooseDies(i);                          // 死亡流程(幼雁沮丧 / apex 斩首混乱)
        eaten = true;
        break;
      }
    }
    if (eaten) continue;

    let vx = V2[2*i] + ax, vy = V2[2*i+1] + ay;
    const sp = Math.hypot(vx, vy);
    if (sp > MAX_SPEED_G * leaderHustle) {
      const lim = MAX_SPEED_G * leaderHustle;
      vx = vx / sp * lim; vy = vy / sp * lim;
    }
    // 每只雁的转向限速(压坡惯性): 速度方向每帧最多转 GOOSE_MAX_TURN
    const curA = Math.atan2(V2[2*i+1], V2[2*i]);
    const newA = Math.atan2(vy, vx);
    const turn = clamp(angDiff(newA, curA), -GOOSE_MAX_TURN, GOOSE_MAX_TURN);
    const fA = curA + turn;
    const fSp = Math.hypot(vx, vy);
    V2[2*i] = Math.cos(fA) * fSp; V2[2*i+1] = Math.sin(fA) * fSp;
    P2[2*i]   = wrapX(xi + V2[2*i]);
    P2[2*i+1] = wrapY(yi + V2[2*i+1]);
  }

  /* ---- 6.6 鹰: Q-learning 决策 ----
   * 状态: 视野内(雁更近? 椋鸟更近?) × (距离近/中/远), 全空 = STATE_NONE
   * 动作: 0 追椋鸟密集处(会冲刺) / 1 追最近的雁(会冲刺)
   *       2 游荡伏击(前移拦截点, 不冲刺) / 3 攻击竞争者(最近的隼, 会冲刺)
   * 奖励: 本帧吃到的分(pending) - 代谢成本(每帧), 帧末按 (s,a)->s' 回流更新
   * 视野内什么都没有时, 先飞向记忆点, 记忆过期才原地盘旋搜索。 */
  {
    const E = eagle;
    if (E.stun > 0) E.stun--;                  // E: 眩晕每帧递减(在动作之前)
    const pc = E.dashT > 0 ? null : perceive(E.x, E.y, E.vx, E.vy, VIEW_EAGLE_R2);
    E.seeing = !!pc;
    if (pc) { E.memX = pc.x; E.memY = pc.y; E.memT = frame; }
    // P2D 威慑锥: 锥内持续失血 —— 走 pending, 让 Q 表把"别闯雁群正面"学进去
    if (inCone(E.x, E.y)) { E.pending -= CONE_EAGLE_DRAIN; stats.coneEagleDrain++; }

    let gi = -1, gd2 = 1e18;                   // 视野内最近的雁
    for (let j = 0; j < M; j++) {
      if (!galive[j]) continue;
      const dx = minImageX(P2[2*j] - E.x), dy = minImageY(P2[2*j+1] - E.y);
      const d2 = dx * dx + dy * dy;
      if (d2 < gd2) { gd2 = d2; gi = j; }
    }
    const gDist = gi >= 0 ? Math.sqrt(gd2) : 1e9;
    const sDist = pc ? Math.hypot(minImageX(pc.x - E.x), minImageY(pc.y - E.y)) : 1e9;

    let ri = -1, rd2 = 1e18;                   // E: 最近的竞争者(隼), 动作 3 的目标
    for (let k = 0; k < falcons.length; k++) {
      const dx = minImageX(falcons[k].x - E.x), dy = minImageY(falcons[k].y - E.y);
      const d2 = dx * dx + dy * dy;
      if (d2 < rd2) { rd2 = d2; ri = k; }
    }

    let state;
    if (gDist === 1e9 && sDist === 1e9) state = STATE_NONE;
    else {
      const prey = (gDist < sDist) ? 1 : 0;
      const dd = Math.min(gDist, sDist);
      const bin = dd < 200 ? 0 : (dd < 500 ? 1 : 2);
      state = prey * 3 + bin;
    }

    let a = 2;                                 // ε-greedy; 无目标时只能游荡
    if (state === STATE_NONE) {
      E.hold = 0;
    } else if (E.hold > 0 && state === E.holdState) {
      a = E.holdAction; E.hold--;              // 决策纪元内保持同一动作
    } else {
      if (rand() < E.eps) a = (rand() * N_ACTIONS) | 0;
      else {
        a = 0;
        for (let k = 1; k < N_ACTIONS; k++)
          if (E.Q[state*N_ACTIONS+k] > E.Q[state*N_ACTIONS+a]) a = k;
      }
      E.hold = EAGLE_DECIDE_EVERY - 1;
      E.holdState = state;
    }
    E.holdAction = a;
    E.curState = state; E.curAction = a;

    let txp = 0, typ = 0, hasT = false;
    if (a === 1 && gi >= 0)      { txp = P2[2*gi];   typ = P2[2*gi+1]; hasT = true; }
    else if (a === 0 && pc && pc.lockIdx >= 0) {   // P1: 采样出的"捞勺中心"
      txp = pc.cx; typ = pc.cy; hasT = true;
    }
    else if (a === 2 && pc) {
      // 游荡伏击: 不直线冲进群体, 而是飞到"群体行进方向前方的拦截点"等着。
      // 中心仍来自采样(P1), 用中心附近代表鸟的速度外推 AMBUSH_LEAD 帧; 伏击不冲刺。
      let lvx = 0, lvy = 0;
      if (pc.bi >= 0) { lvx = V[2*pc.bi]; lvy = V[2*pc.bi+1]; }
      txp = wrapX(pc.cx + lvx * AMBUSH_LEAD);
      typ = wrapY(pc.cy + lvy * AMBUSH_LEAD);
      hasT = true;
    }
    else if (a === 3 && ri >= 0) { txp = falcons[ri].x; typ = falcons[ri].y; hasT = true; }
    else if (freshMem(E))        { txp = E.memX;     typ = E.memY;     hasT = true; }

    const dx = hasT ? minImageX(txp - E.x) : 0;
    const dy = hasT ? minImageY(typ - E.y) : 0;
    const dth = hasT ? Math.hypot(dx, dy) : 0;

    if (E.cool > 0) E.cool--;
    if (E.dashT > 0) {                         // 冲刺惯性: 期间不转向
      E.dashT--;
      if (E.dashT === 0) {                     // 冲刺结束
        if (E.dashEats === 0) {                // A: 空冲刺: 扣分并流进 Q 学习
          E.score -= MISS_PENALTY_EAGLE;
          E.pending -= MISS_PENALTY_EAGLE;
          stats.missEagle++;
        }
        E.lockIdx = -1;                        // D: 解除锁定
      }
    }
    else if (E.stun <= 0 && a !== 2 && hasT && dth < EAGLE_DASH_TRIGGER && dth > 1 && E.cool === 0) {
      const curA = Math.atan2(E.vy, E.vx), tgtA = Math.atan2(dy, dx);
      if (Math.abs(angDiff(tgtA, curA)) <= DASH_MAX_ANGLE) {
        E.dashT = EAGLE_DASH_FRAMES; E.cool = EAGLE_DASH_CD;
        E.dashEats = 0;
        // D: 只有"追椋鸟"才是锁定式迷惑(P1: 锁采样中心那 40px 内最近的那只);
        //    追雁/攻隼不锁(雁大不存在迷惑, 对手不是"吓懵"的对象)
        E.lockIdx = (DASH_LOCK && a === 0 && pc && pc.lockIdx >= 0) ? pc.lockIdx : -1;
        if (E.lockIdx >= 0) stats.lockedDashes++;
      }
    }
    // C: 被围攻拖慢 + E: 眩晕减速(都只作用于非冲刺状态)
    const slow = Math.max(MOB_SLOW_EAGLE, 1 - E.mob * 0.005) * (E.stun > 0 ? 0.65 : 1);
    const spd = E.dashT > 0 ? EAGLE_DASH : EAGLE_CRUISE * slow;
    if (E.dashT === 0) {
      if (hasT && dth > 1) { E.vx += dx / dth * 0.25; E.vy += dy / dth * 0.25; }
      else {                                   // 纯搜索: 缓慢转向, 画大圈找人
        const ca = Math.cos(0.05), sa = Math.sin(0.05);
        const nvx = E.vx * ca + E.vy * sa, nvy = -E.vx * sa + E.vy * ca;
        E.vx = nvx; E.vy = nvy;
      }
    }
    applyBoundary(E, 2);                       // 开阔世界别飞丢了
    const vs = Math.hypot(E.vx, E.vy) || 1;
    E.vx = E.vx / vs * spd; E.vy = E.vy / vs * spd;
    E.x = wrapX(E.x + E.vx); E.y = wrapY(E.y + E.vy);
    trailPush(E, E.x, E.y);

    // Q 更新: 把本帧吃到的分记到上一步 (状态, 动作) 头上
    E.pending -= METAB_EAGLE;                  // B: 代谢成本先进 pending, 学的才是净期望
    const reward = E.pending; E.pending = 0;
    let bn = E.Q[state*N_ACTIONS];
    for (let k = 1; k < N_ACTIONS; k++)
      if (E.Q[state*N_ACTIONS+k] > bn) bn = E.Q[state*N_ACTIONS+k];
    const q = E.lastState * N_ACTIONS + E.lastAction;
    E.Q[q] += Q_ALPHA * (reward + Q_GAMMA * bn - E.Q[q]);
    E.lastState = state; E.lastAction = a;
    E.eps = Math.max(EPS_MIN, E.eps * EPS_DECAY);
  }

  /* ---- 6.7 隼的策略进化: (1+1)-ES 期界结算(在隼行动之前跑, 新期参数当帧生效) ----
   * 原则: 制度(能量轮换/拦截位/交棒/驱逐)不学, 学的是制度内的操作参数 ——
   * 鹰学价值(Q-learning), 隼学策略(进化), 两种学习哲学同框。
   * 双缓冲: periods 为偶数的期评估"当前参数", 奇数的期评估"当前 + N(0,σ) 截断"的候选;
   * 期末配对比较(期长 3600 帧, 因为适应度噪声大), 候选更好就采纳(p=cand, gen++),
   * 否则回滚。适应度 = 期内净得分:
   *     fit = Δscore - ES_MISS_WEIGHT·Σ落空罚款 - ES_METAB_WEIGHT·Σ代谢成本
   * 无价值函数、无状态抽象, 纯爬山 —— 允许学成"永不出手"的局部最优(适应度≈0),
   * 那是诚实的失败模式, 不是 bug。p6 线性冲刺策略留作扩展位(见交接说明)。 */
  for (let fi = 0; fi < falcons.length; fi++) {
    const f = falcons[fi], pol = f.policy;
    if (pol.t === 0) {                           // 期初: 选这一期跑谁
      if (pol.periods % 2 === 1) {
        const c = new Float64Array(ES_NP);
        for (let k = 0; k < ES_NP; k++) c[k] = clamp(pol.p[k] + randn() * ES_SIGMA, 0, 1);
        pol.evalP = c; pol.isCand = true;        // 候选(每只隼各抽各的, 随机源不共享)
      } else {
        pol.evalP = pol.p; pol.isCand = false;   // 保留当前参数
      }
      pol.score0 = f.score; pol.miss = 0; pol.metab = 0;
    }
    pol.t++;
    if (pol.t >= (esEvalOverride || ES_EVAL_FRAMES)) {   // 期末: 配对比较 + 采纳/回滚
      const fit = (f.score - pol.score0) - ES_MISS_WEIGHT * pol.miss - ES_METAB_WEIGHT * pol.metab;
      pol.lastFit = fit;
      if (pol.isCand) {
        if (fit > pol.baseFit) { pol.p.set(pol.evalP); pol.gen++; stats.esAdopt++; }
      } else {
        pol.baseFit = fit;
      }
      pol.periods++; stats.esPeriods++;
      pol.t = 0; pol.evalP = null;
    }
  }

  /* ---- 6.8 隼: 能量轮换 + 拦截位配合 + (围攻/眩晕/驱逐) + 策略参数(P3) ---- */
  for (let fi = 0; fi < falcons.length; fi++) {
    const f = falcons[fi];
    const isActive = fi === activeId;
    if (f.stun > 0) f.stun--;                  // E: 眩晕每帧递减
    if (f.clashCd > 0) f.clashCd--;
    const PV_ = policyVals(f);                 // P3: 本评估期生效的策略参数
    const coneHit = inCone(f.x, f.y);          // P2D: 闯进雁群正面 -> 失血
    if (coneHit) stats.coneFalconFrames++;

    const pc = f.dashT > 0 ? null : perceive(f.x, f.y, f.vx, f.vy, VIEW_FALCON_R2);
    f.seeing = !!pc;
    if (pc) { f.memX = pc.x; f.memY = pc.y; f.memT = frame; }

    if (f.dashT > 0) {
      // 冲刺: 速度锁定不调整; B 代谢 + C 围攻者越多烧得越快(容忍度 p5 由 ES 学)
      f.dashT--;
      const coneMult = coneHit ? CONE_FALCON_DASH_MULT : 1;      // P2D: 锥内冲刺能耗 ×2
      const drain = (DASH_DRAIN * (1 + f.mob * MOB_DRAIN_PER * PV_.mobTol) + METAB_DASH) * coneMult;
      f.energy = Math.max(0, f.energy - drain);
      f.policy.metab += drain;                 // P3: 代谢成本计入适应度
      applyBoundary(f, 2);
      // C: 被围到临界值 -> 冲刺被硬生生打断, 走下面正常的落空罚款流程
      if (f.mob >= MOB_PENALTY_N) { f.dashT = 0; stats.mobInterrupts++; }
      const ps = Math.hypot(f.vx, f.vy);
      if (ps > 1e-9) { f.vx = f.vx / ps * DASH_SPEED; f.vy = f.vy / ps * DASH_SPEED; }
      if (f.dashT === 0) {                     // 冲刺结束
        if (f.dashEats === 0) {                // A: 空冲刺扣分
          f.score -= MISS_PENALTY_FALCON;
          f.policy.miss += MISS_PENALTY_FALCON;   // P3: 落空罚款计入适应度
          stats.missFalcon++;
        }
        // P1 验收取样: 冲刺终点 80px 内还有多少可见鸟(拣软处下口是否自发产生)
        const pe = perceive(f.x, f.y, f.vx, f.vy, 80 * 80);
        stats.dashEnds++;
        stats.dashEndBirds += pe ? pe.n : 0;
        if (f.dashS > stats.dashMaxEats) stats.dashMaxEats = f.dashS;
        if (f.lockIdx >= 0) {                  // 锁定冲刺(捕猎)
          if (f.dashS > stats.dashMaxEatsLocked) stats.dashMaxEatsLocked = f.dashS;
        } else if (f.dashS > stats.dashMaxEatsFree) stats.dashMaxEatsFree = f.dashS;
        // P2D 验收: 终点与 V 尖的关系(锥是否真的把冲刺挤开了)
        if (galiveCount > 0) {
          const dA = Math.hypot(minImageX(f.x - Lx), minImageY(f.y - Ly));
          stats.dashEndApexSum += dA;
          if (dA < 200) stats.dashEndApexNear++;
          if (inCone(f.x, f.y)) stats.dashEndInCone++;
        }
        f.circleT = CIRCLE_FRAMES;
        f.lockIdx = -1;                        // D: 解除锁定
      }
    } else {
      // C 围攻 + E 眩晕: 都只拖慢非冲刺状态
      const slow = Math.max(MOB_SLOW_FALCON, 1 - f.mob * 0.01) * (f.stun > 0 ? 0.6 : 1);

      // 盘旋: 缓慢绕圈(所有非冲刺状态都叠这一层转向)
      const ca = Math.cos(CIRCLE_TURN), sa = Math.sin(CIRCLE_TURN);
      const nvx = f.vx * ca + f.vy * sa;
      const nvy = -f.vx * sa + f.vy * ca;
      f.vx = nvx; f.vy = nvy;

      if (isActive) {
        // 追猎方: 视野内朝群体质心修正, 看不见就飞向记忆点
        if (pc) {
          const dfx = minImageX(pc.x - f.x), dfy = minImageY(pc.y - f.y);
          const df = Math.hypot(dfx, dfy);
          if (df > 200) { f.vx += dfx / df * 0.6; f.vy += dfy / df * 0.6; }
        } else if (freshMem(f)) {
          const mx = minImageX(f.memX - f.x), my = minImageY(f.memY - f.y);
          const md = Math.hypot(mx, my);
          if (md > 1) { f.vx += mx / md * 0.6; f.vy += my / md * 0.6; }
        }
        applyBoundary(f, 2);
        const ps = Math.hypot(f.vx, f.vy);
        const cspd = CIRCLE_SPEED * slow;
        if (ps > 1e-9) { f.vx = f.vx / ps * cspd; f.vy = f.vy / ps * cspd; }

        f.energy = Math.min(ENERGY_MAX, f.energy + ACTIVE_REGEN - METAB_CIRCLE);
        f.policy.metab += METAB_CIRCLE;
        f.circleT--;

        // 能量低(p2) -> 退出追猎, 让设伏位的另一只立刻接棒
        if (f.energy < PV_.energyLow) {
          activeId = 1 - fi;
          falcons[activeId].circleT = 0;
          continue;
        }
        // 盘旋期满 + 能量够(p1) + 不晕 + 采样到可用目标 -> 冲刺(60° 约束, D 锁定)
        if (f.circleT <= 0 && f.energy >= PV_.dashMin && f.stun <= 0 && pc) {
          if (pc.cx < 0) { stats.noLockAborts++; f.circleT = 4; }   // P1: 40px 内无鸟 -> 放弃(隔几帧再采样)
          else if (!tryDash(f, pc.cx, pc.cy, pc.lockIdx)) f.circleT = 8;
        }
      } else {
        // 设伏方: 不算全局鸟群, 只用自己视野里的质心推拦截位 ——
        // 鸟群总朝远离追猎方的方向逃, 逃逸走廊的尽头就是设伏点(p4 外推比例可学)。
        let travel = false, tx = 0, ty = 0, d = 0;
        if (pc) {
          const af = falcons[activeId];
          const ax0 = minImageX(pc.x - af.x), ay0 = minImageY(pc.y - af.y);
          const ipx = wrapX(pc.x + ax0 * PV_.interceptAhead);
          const ipy = wrapY(pc.y + ay0 * PV_.interceptAhead);
          tx = minImageX(ipx - f.x); ty = minImageY(ipy - f.y);
          d = Math.hypot(tx, ty);
          travel = d > INTERCEPT_MIN;
        } else if (freshMem(f)) {
          tx = minImageX(f.memX - f.x); ty = minImageY(f.memY - f.y);
          d = Math.hypot(tx, ty);
          travel = d > INTERCEPT_MIN;
        }
        if (travel && d > 1) {
          // 未到拦截位: 高速赶路(A 版: 设伏方速度 7.5 > 盘旋 5.0)
          f.vx += clamp(tx / d * REST_SPEED - f.vx, -0.8, 0.8);
          f.vy += clamp(ty / d * REST_SPEED - f.vy, -0.8, 0.8);
        } else if (pc && d > 1) {
          f.vx += tx / d * INTERCEPT_GAIN;     // 已到位: 小幅贴向设伏点
          f.vy += ty / d * INTERCEPT_GAIN;
        }
        applyBoundary(f, 2);
        const ps = Math.hypot(f.vx, f.vy);
        const spd = (travel ? REST_SPEED : CIRCLE_SPEED) * slow;
        if (ps > 1e-9) { f.vx = f.vx / ps * spd; f.vy = f.vy / ps * spd; }

        f.energy = Math.min(ENERGY_MAX, f.energy + REST_REGEN - METAB_CIRCLE);
        f.policy.metab += METAB_CIRCLE;
        f.circleT--;

        // 偶发双打(p3): 设伏方能量很满且采样到可用目标时, 不等接棒直接出手
        if (f.circleT <= 0 && f.energy >= PV_.doubleTeam && f.stun <= 0 && pc) {
          if (pc.cx < 0) { stats.noLockAborts++; f.circleT = 4; }
          else if (!tryDash(f, pc.cx, pc.cy, pc.lockIdx)) f.circleT = 10;
        }
      }

      // E 驱逐冲刺: 优先级低于捕猎 —— 附近有对手(另一只隼或鹰)且能量够就撞上去,
      // 撞上后由 6.9 的互攻判定决胜负。没有猎物可追时这就是唯一出路。
      if (f.dashT === 0 && f.stun <= 0 && f.circleT <= 0 && f.energy >= PV_.dashMin) {
        let rv = null, rvd2 = 1e18;
        for (const r of raptors) {
          if (r === f) continue;
          const dx = minImageX(r.x - f.x), dy = minImageY(r.y - f.y);
          const d2 = dx * dx + dy * dy;
          if (d2 < rvd2) { rvd2 = d2; rv = r; }
        }
        if (rv && rvd2 < RIVAL_R * RIVAL_R) {
          if (!tryDash(f, rv.x, rv.y, -1)) f.circleT = 6;
        }
      }
    }
    f.x = wrapX(f.x + f.vx);
    f.y = wrapY(f.y + f.vy);
    trailPush(f, f.x, f.y);
  }

  /* ---- 6.9 猎物补充 ---- */
  if (frame % REFILL_CHECK === 0) maybeRefill();

  /* ---- 6.10 猛禽互攻(驱逐制): 三只两两判定 ----
   * 实力 = 冲刺中 +2 / 鹰体型 +1.5 / 眩晕 -3 / 噪声;
   * 败方被击退 + 眩晕 + 扣分 + 被抢走剩余分数的 STEAL_FRAC; 胜方拿走抢来的分。
   * 加了 CLASH_CD 冷却: 否则贴在一起的两只猛禽会逐帧互相扣分, 分数瞬间崩掉。 */
  for (let i = 0; i < raptors.length; i++) {
    for (let j = i + 1; j < raptors.length; j++) {
      const A = raptors[i], B = raptors[j];
      if (A.clashCd > 0 || B.clashCd > 0) continue;
      const dx = minImageX(B.x - A.x), dy = minImageY(B.y - A.y);
      const d2 = dx * dx + dy * dy;
      if (d2 >= CLASH_R * CLASH_R || d2 < 1e-6) continue;

      const sA = (A.dashT > 0 ? 2 : 0) + (A === eagle ? 1.5 : 0) + (A.stun > 0 ? -3 : 0) + rand();
      const sB = (B.dashT > 0 ? 2 : 0) + (B === eagle ? 1.5 : 0) + (B.stun > 0 ? -3 : 0) + rand();
      const win = sA >= sB ? A : B, lose = win === A ? B : A;
      const md = Math.sqrt(d2);
      const nx = dx / md, ny = dy / md;              // A -> B 的单位向量

      // 败方被击退(位置 + 速度都推一把), 眩晕, 扣分, 被抢分
      lose.x = wrapX(lose.x + (lose === B ? nx : -nx) * CLASH_KNOCK);
      lose.y = wrapY(lose.y + (lose === B ? ny : -ny) * CLASH_KNOCK);
      lose.vx = (lose === B ? nx : -nx) * 4;
      lose.vy = (lose === B ? ny : -ny) * 4;
      lose.stun = CLASH_STUN;
      if (lose.dashT > 0) {                          // 被打断的冲刺同样走落空罚款流程
        if (lose.dashEats === 0) {
          if (lose === eagle) {
            lose.score -= MISS_PENALTY_EAGLE; lose.pending -= MISS_PENALTY_EAGLE; stats.missEagle++;
          } else {
            lose.score -= MISS_PENALTY_FALCON; stats.missFalcon++;
          }
        }
        if (lose !== eagle) lose.circleT = CIRCLE_FRAMES;
      }
      lose.dashT = 0; lose.lockIdx = -1;
      lose.score -= CLASH_COST;
      const stolen = lose.score > 0 ? Math.floor(lose.score * STEAL_FRAC) : 0;
      lose.score -= stolen;
      win.score += stolen;
      win.dashEats++;                                // 胜方免空冲刺罚款
      // 鹰的奖惩要走 pending, 否则 Q 学习看不见互攻这件事
      if (win === eagle) eagle.pending += stolen;
      if (lose === eagle) eagle.pending -= CLASH_COST;
      A.clashCd = CLASH_CD; B.clashCd = CLASH_CD;
      stats.clashes++;
      if (stolen > 0) stats.steals++;
    }
  }
}

/* ============================================================================
 * WebGPU 计算路径(第 6 轮, 依据 webgpu-migration-plan.md)
 * ----------------------------------------------------------------------------
 * 上 GPU: 6.2 拓扑邻居力 + 6.4 积分/被捕食/逃逸/围攻/惊飞(实测占 step 耗时 ~90%,
 *         且全是数据并行)。留 CPU: 6.1 网格(perceive 要用, 仅 ~5%)、6.3 迁徙
 *         (结果作 uniform 上传)、6.5-6.10 全部"智能"。
 * 每帧数据流: 下行 = 128B uniform(3 只猛禽 pos/vel/dashT/lockIdx + n/k/mig);
 *         上行 = staging 双缓冲(全量 P/V/PV/flags 约 560KB @N=20000 + 256B 帧摘要,
 *         mapAsync 异步取上一帧, 渲染/感知消费回读后的数组, AI 只消费摘要)。
 * 环面(torus)模式自动回退 CPU(圆均值质心未上 GPU); 任何帧失败永久回退 CPU。
 * ========================================================================== */
let computeMode = "auto";                  // ?compute=cpu|gpu|auto(默认 auto)
try { const m = /[?&]compute=(cpu|gpu|auto)/.exec(window.location.search); if (m) computeMode = m[1]; } catch (e) { /* headless */ }
let gpuActive = false;                     // 本 rAF 是否走 WebGPU(每帧重算)
let gpuBusy = false;                       // gpuFrameCore 重入锁(每 rAF 至多一帧在飞)
let gpuInitTried = false;

function gpuSyncBirds() {                  // CPU 侧改了 P/V/PV/alive 后同步上 GPU
  if (window.ComputeBackend && window.ComputeBackend.ready && !TORUS) window.ComputeBackend.uploadBirds();
}

function applyGpuMeta(meta) {              // 帧摘要 -> 猛禽计分/围攻计数/死亡涟漪
  if (!meta || meta.n !== N) return;       // N 中途变更: 丢弃过期回读(initStarlings 已全量上传)
  aliveCount = meta.cnt;
  cxm = meta.cx; cym = meta.cy;
  for (let t = 0; t < 2; t++) {
    const f = falcons[t], K = meta.kills[t];
    if (K) { f.score += K; f.kills += K; stats.kills += K; }
    f.dashEats += meta.dashEats[t];
    f.dashS = (f.dashS || 0) + meta.dashEats[t];
    f.mob = meta.mob[t];
  }
  const K2 = meta.kills[2];
  if (K2) { eagle.score += K2; eagle.kills += K2; eagle.pending += K2; eagle.dietStarling += K2; stats.kills += K2; }
  eagle.dashEats += meta.dashEats[2];
  eagle.dashS = (eagle.dashS || 0) + meta.dashEats[2];
  eagle.mob = meta.mob[2];
  const ev = meta.evIdx;
  for (let e = 0; e < meta.events; e++) {
    const i = ev[e];
    if (i < N) addDeath(P[2*i], P[2*i+1]); // 死亡涟漪: 回读位置即临终位置
  }
}

async function gpuFrameCore() {
  frame++; stats.steps++;
  const t0 = performance.now();
  doMigrate();                             // 6.3: 质心用上一帧摘要(一帧延迟, 无感知影响)
  window.ComputeBackend.submit();                 // 6.2 + 6.4 上 GPU; 与下方 CPU 工作流水线重叠
  const meta = await window.ComputeBackend.readbackPrev();
  rebuildGrid();                           // 6.1 留 CPU(清 mob 计数 + perceive 预筛)
  applyGpuMeta(meta);                      // 击杀/冲刺进食/围攻/存活数/死亡涟漪
  stepTail();                              // 6.5-6.10 留 CPU
  perf.stepMs = perf.stepMs * 0.9 + (performance.now() - t0) * 0.1;
}

function stepGPUFrame() {                  // rAF 入口: 上一帧未完工就让位(不堆积)
  if (gpuBusy) return false;
  gpuBusy = true;
  gpuFrameCore()
    .catch(err => { console.warn("WebGPU 帧失败, 永久回退 CPU:", err); computeMode = "cpu"; })
    .finally(() => { gpuBusy = false; });
  return true;
}

/* ---- Phase 0 回归验收: 固定种子 + 猛禽 parked + 关闭补充, CPU/GPU 各跑一遍 ---- */
function pearson(a, b) {
  const n = Math.min(a.length, b.length);
  if (n < 4) return 1;
  let ma = 0, mb = 0;
  for (let i = 0; i < n; i++) { ma += a[i]; mb += b[i]; }
  ma /= n; mb /= n;
  let sab = 0, saa = 0, sbb = 0;
  for (let i = 0; i < n; i++) {
    const da = a[i] - ma, db = b[i] - mb;
    sab += da * db; saa += da * da; sbb += db * db;
  }
  return saa > 0 && sbb > 0 ? sab / Math.sqrt(saa * sbb) : 0;
}

async function goldenRun(frames, mode) {   // mode: "cpu" | "gpu"(GPU 须已就绪)
  if (mode === "gpu" && !(window.ComputeBackend && window.ComputeBackend.ready))
    throw new Error("WebGPU 未就绪, 无法跑 GPU golden");
  const keepRefill = refillOn;
  refillOn = false;                        // 补充会复活死鸟, 污染等价对比
  if (TORUS) setGeometry(false);           // GPU 路径仅开阔世界
  setSeed(0x5EED);
  resetFlocks();
  resetPredators();
  const park = () => {
    for (const r of raptors) {
      r.x = -PAD - 40; r.y = -PAD - 40; r.vx = 0; r.vy = 0;
      r.dashT = 0; r.energy = ENERGY_MAX; r.memT = -1e9; r.seeing = false;
    }
  };
  const trail = [];
  for (let f = 0; f < frames; f++) {
    park();                                // 每帧钉住猛禽: 只验鸟侧物理等价
    if (mode === "gpu") await gpuFrameCore();
    else step();
    if (f % 10 === 0) trail.push(cxm, cym);
  }
  refillOn = keepRefill;
  let mvx = 0, mvy = 0, c = 0, sx = 0, sy = 0;
  for (let i = 0; i < N; i++) {
    if (!alive[i]) continue;
    mvx += V[2*i]; mvy += V[2*i+1]; sx += P[2*i]; sy += P[2*i+1]; c++;
  }
  const polar = c ? Math.hypot(mvx, mvy) / c / MAX_SPEED : 0;
  const mx = c ? sx / c : 0, my = c ? sy / c : 0;
  let spread = 0;
  for (let i = 0; i < N; i++) {
    if (!alive[i]) continue;
    spread += Math.hypot(P[2*i] - mx, P[2*i+1] - my);
  }
  return { kills: stats.kills, alive: aliveCount, trail,
           polar: +polar.toFixed(4), spread: +(c ? spread / c : 0).toFixed(1) };
}

// 渲染侧零分配: 虚线样式数组与辉光精灵都预生成一次, 不再每帧 new
const EMPTY_DASH = [];
const DASH_VISION = [4, 10], DASH_MEM = [5, 7], DASH_AXIS = [3, 9], DASH_FORM = [6, 8], DASH_MINI = [3, 4];
const glowSprites = new Map();                // 颜色 -> 64×64 预渲染径向辉光
function glowSprite(C) {
  let s = glowSprites.get(C);
  if (s) return s;
  const c = document.createElement("canvas");
  c.width = 64; c.height = 64;
  const g2 = c.getContext("2d");
  const grad = g2.createRadialGradient(32, 32, 0, 32, 32, 32);
  grad.addColorStop(0, `rgba(${C}, 1)`);
  grad.addColorStop(1, `rgba(${C}, 0)`);
  g2.fillStyle = grad;
  g2.fillRect(0, 0, 64, 64);
  glowSprites.set(C, c);
  return c;
}

function drawArrow(x, y, vx, vy, sTip, sBack, sWing) {
  // 通用 ">": 尖端在速度方向 sTip 处, 批量塞进同一条 path 一次描边
  const sp = Math.hypot(vx, vy) || 1;
  const ux = vx / sp, uy = vy / sp, qx = -uy, qy = ux;
  const tx = x + ux * sTip, ty = y + uy * sTip;
  const bx = x - ux * sBack, by = y - uy * sBack;
  ctx.moveTo(bx + qx * sWing, by + qy * sWing);
  ctx.lineTo(tx, ty);
  ctx.lineTo(bx - qx * sWing, by - qy * sWing);
}

// 拖尾环形缓冲: 定容 Float32Array + 头指针, 推一个点 O(1), 永不分配
function trailPush(o, x, y) {
  const L = TRAIL_LEN;
  if (o.trailN < L) {
    const idx = (o.trailHead + o.trailN) * 2;
    o.trailBuf[idx] = x; o.trailBuf[idx + 1] = y; o.trailN++;
  } else {
    o.trailBuf[o.trailHead * 2] = x; o.trailBuf[o.trailHead * 2 + 1] = y;
    o.trailHead = (o.trailHead + 1) % L;
  }
}

// 拖尾颜色字符串缓存: 按(颜色, 透明度档位)预生成, 避免每段一次模板字符串
const TRAIL_BUCKETS = 12;
const trailColorCache = new Map();
function trailColor(C, a) {
  let bucket = (a * TRAIL_BUCKETS) | 0;
  if (bucket < 0) bucket = 0; else if (bucket >= TRAIL_BUCKETS) bucket = TRAIL_BUCKETS - 1;
  const key = C + "|" + bucket;
  let s = trailColorCache.get(key);
  if (s === undefined) {
    s = `rgba(${C}, ${((bucket + 0.5) / TRAIL_BUCKETS).toFixed(3)})`;
    trailColorCache.set(key, s);
  }
  return s;
}

// 猛禽的公共画法(隼/鹰共用): 拖尾、冲击波、视野圈、搜索记忆线、辉光、本体、能量条
function drawHunter(h, dash, opt) {
  const C = h.color;

  const n = h.trailN;
  if (n >= 2) {                                // 环形缓冲里按时间顺序连点
    const L = TRAIL_LEN;
    ctx.lineCap = "round";
    let px = h.trailBuf[h.trailHead * 2], py = h.trailBuf[h.trailHead * 2 + 1];
    for (let k = 1; k < n; k++) {
      const idx = ((h.trailHead + k) % L) * 2;
      const x = h.trailBuf[idx], y = h.trailBuf[idx + 1];
      const dx = x - px, dy = y - py;
      if (dx * dx + dy * dy <= 1600) {         // 穿屏/瞬移跳变不连线
        ctx.strokeStyle = trailColor(C, opt.trailA * (k + 1) / n);
        ctx.lineWidth = dash ? 2.5 : 1.5;
        if (dash) { ctx.shadowColor = `rgb(${C})`; ctx.shadowBlur = 14; }
        ctx.beginPath();
        ctx.moveTo(px, py);
        ctx.lineTo(x, y);
        ctx.stroke();
      }
      px = x; py = y;
    }
    ctx.shadowBlur = 0;
  }

  if (h.shock) {                               // 冲刺冲击波
    h.shock.t++;
    const r = h.shock.t * 9, alpha = 1 - h.shock.t / 35;
    if (alpha <= 0) h.shock = null;
    else {
      ctx.strokeStyle = `rgba(${C}, ${alpha * 0.8})`;
      ctx.lineWidth = 2.5;
      ctx.shadowColor = `rgb(${C})`; ctx.shadowBlur = 20;
      ctx.beginPath();
      ctx.arc(h.shock.x, h.shock.y, r, 0, TAU);
      ctx.stroke();
      ctx.shadowBlur = 0;
    }
  }

  if (showVision && opt.visionR) {             // 视野圈: 直观验证"无全局视野"
    ctx.save();
    ctx.setLineDash(DASH_VISION);
    ctx.strokeStyle = `rgba(${C}, 0.10)`;
    ctx.lineWidth = 1;
    ctx.beginPath();
    ctx.arc(h.x, h.y, opt.visionR, 0, TAU);
    ctx.stroke();
    ctx.restore();
  }

  if (!h.seeing && freshMem(h)) {              // 搜索中: 朝记忆点飞的虚线
    ctx.save();
    ctx.setLineDash(DASH_MEM);
    ctx.strokeStyle = `rgba(${C}, 0.20)`;
    ctx.lineWidth = 1;
    ctx.beginPath();
    ctx.moveTo(h.x, h.y);
    ctx.lineTo(h.x + minImageX(h.memX - h.x), h.y + minImageY(h.memY - h.y));
    ctx.stroke();
    ctx.restore();
  }

  const glowR = dash ? opt.glowDash : opt.glow;
  ctx.globalAlpha = dash ? 0.55 : opt.glowA;   // 预渲染辉光精灵, 不再每帧建渐变
  ctx.drawImage(glowSprite(C), h.x - glowR, h.y - glowR, glowR * 2, glowR * 2);
  ctx.globalAlpha = 1;

  ctx.strokeStyle = `rgb(${C})`;
  ctx.lineWidth = dash ? 3 : 2;
  ctx.shadowColor = `rgb(${C})`;
  ctx.shadowBlur = dash ? 24 : 12;
  ctx.beginPath();
  drawArrow(h.x, h.y, h.vx, h.vy, dash ? opt.size[0] : opt.size[1], opt.size[2], opt.size[2]);
  ctx.stroke();
  ctx.shadowBlur = 0;
}

/* ---- 渲染分发: 默认走 WebGL 实例化后端, ?render=2d / 无 WebGL 时回退 Canvas2D ---- */
function render() {
  const t0 = performance.now();
  if (glReady) {
    try {
      window.GLBackend.draw(glScene);          // 底层: 残影 ping-pong + 两批实例化鸟
      glUsed = true;
    } catch (e) {
      glReady = false;                         // GL 出任何错就当场降级, 不让画面黑掉
      perf.gl = false;
      console.warn("WebGL 后端异常, 回退 Canvas2D:", e && e.message);
    }
  }
  if (glReady) {
    renderOverlay();                           // 顶层 2D: 猛禽/标签/涟漪/小地图/锥
  } else {
    render2D();                                // 原始 Canvas2D 全路径
  }
  perf.renderMs = perf.renderMs * 0.9 + (performance.now() - t0) * 0.1;
  perf.gl = glReady;
}

// 小地图: 开阔世界专用 —— 视口只是一部分, 必须有个全局视图看鸟群跑哪去了

  for (let i = 0; i < N; i += 6) {

  }
  // P2: 领队位置 + 威慑锥方向(小地图上最容易看出"雁群在躲谁")
  if (galiveCount > 0) {
    ctx.fillStyle = "rgba(120,190,255,0.9)";
    ctx.fillRect(mx(Lx) - 2, my(Ly) - 2, 4, 4);
    ctx.strokeStyle = "rgba(120,190,255,0.5)";
    ctx.beginPath();
    ctx.moveTo(mx(Lx), my(Ly));
    ctx.lineTo(mx(Lx + Math.cos(leaderAng) * 60), my(Ly + Math.sin(leaderAng) * 60));
    ctx.stroke();
  }

  ctx.fillStyle = "rgba(150,155,168,0.75)";
  ctx.font = "10px 'Microsoft YaHei', sans-serif";
  ctx.fillText(`开阔世界 ${W + 2 * PAD}×${H + 2 * PAD}（虚线 = 视口）`, ox + 2, oy - 4);
  ctx.restore();
}

// 鹰脑 Q 表热力图(小 canvas, 6Hz 刷新足够)
function renderQ() {
  const w = qcv.width, h = qcv.height;
  qctx.clearRect(0, 0, w, h);
  qctx.fillStyle = "rgba(255,255,255,0.03)";
  qctx.fillRect(0, 0, w, h);

  const labelW = 30, headH = 15;
  const cw = (w - labelW) / N_ACTIONS, ch = (h - headH) / N_STATES;
  const rows = ["椋近", "椋中", "椋远", "雁近", "雁中", "雁远", "全空"];
  const cols = ["追椋", "追雁", "伏击", "攻隼"];

  let maxQ = 1e-6;
  for (let s = 0; s < N_STATES; s++)
    for (let a = 0; a < N_ACTIONS; a++)
      maxQ = Math.max(maxQ, eagle.Q[s * N_ACTIONS + a]);

  qctx.font = "9px 'Microsoft YaHei', sans-serif";
  qctx.textBaseline = "middle";
  for (let a = 0; a < N_ACTIONS; a++) {
    qctx.fillStyle = "rgba(180,185,200,0.8)";
    qctx.fillText(cols[a], labelW + a * cw + cw / 2 - 11, headH / 2);
  }
  for (let s = 0; s < N_STATES; s++) {
    const y = headH + s * ch;
    const cur = (s === eagle.curState);
    // 该状态下的贪心动作(顶部亮线), 一眼看出"学会了什么"。
    // 若第一名与第二名差距 <5%, 亮线画成半透明 —— 表示"还没学出偏好",
    // 免得用户以为热力图在闪就是坏了(雁近行常年就是这种势均力敌)。
    let ga = 0, gb = -1;
    for (let a = 1; a < N_ACTIONS; a++) {
      const v = eagle.Q[s * N_ACTIONS + a], cur2 = eagle.Q[s * N_ACTIONS + ga];
      if (v > cur2) { gb = ga; ga = a; } else if (a !== ga && (gb < 0 || v > eagle.Q[s * N_ACTIONS + gb])) gb = a;
    }
    const best = eagle.Q[s * N_ACTIONS + ga];
    const second = gb >= 0 ? eagle.Q[s * N_ACTIONS + gb] : -Infinity;
    const tie = !(best > second) || (best - second) < 0.05 * Math.max(maxQ, 1e-6);
    const hasValue = maxQ > 1e-6 && best > 1e-6;
    qctx.fillStyle = cur ? "rgba(255,220,140,0.95)" : "rgba(150,155,170,0.65)";
    qctx.fillText(rows[s], 2, y + ch / 2);
    for (let a = 0; a < N_ACTIONS; a++) {
      const x = labelW + a * cw;
      const v = Math.max(0, eagle.Q[s * N_ACTIONS + a]) / maxQ;   // 0..1 亮度
      qctx.fillStyle = `hsl(${210 - 165 * v}, ${35 + 40 * v}%, ${10 + 38 * v}%)`;
      qctx.fillRect(x + 1, y + 1, cw - 2, ch - 2);
      if (s !== STATE_NONE && hasValue) {                    // 贪心动作: 顶部一条亮线
        if (a === ga) {
          qctx.fillStyle = tie ? "rgba(255,235,170,0.35)" : "rgba(255,235,170,0.9)";
          qctx.fillRect(x + 3, y + 2, cw - 6, 1.5);
        }
      }
      if (cur && a === eagle.curAction) {                    // 当前决策高亮
        qctx.strokeStyle = "rgba(255,255,255,0.95)";
        qctx.lineWidth = 1.5;
        qctx.strokeRect(x + 1.5, y + 1.5, cw - 3, ch - 3);
      }
    }
  }
}

let stepAcc = 0, loopErrShown = false;
function loop() {                    // 外壳: 单帧异常不得弄死动画循环
  try { loopBody(); }
  catch (e) {
    if (!loopErrShown) { loopErrShown = true; console.error("主循环异常(已捕获, 循环继续):", e); }
  }
  requestAnimationFrame(loop);
}

function loopBody() {
  const now = performance.now();
  fpsEma = fpsEma * 0.95 + (1000 / Math.max(now - lastT, 1)) * 0.05;
  lastT = now;
  tryInitGL();                                 // 首帧再初始化 GL(那时第二个 script 已解析)

  // WebGPU 计算后端: 首帧尝试初始化(?compute=cpu 则永不尝试)
  // 注意: 后端脚本若未加载, window.ComputeBackend 是 undefined —— 必须先判空再调用,
  // 否则每帧抛 ReferenceError, loop() 末尾的 requestAnimationFrame 不执行 => 页面卡死。
  const CB = window.ComputeBackend;
  if (computeMode !== "cpu" && !gpuInitTried && CB) {
    gpuInitTried = true;
    CB.init().then(ok => {
      if (ok) console.info("WebGPU 计算后端就绪: 椋鸟物理(6.2 邻居力 + 6.4 积分)已上 GPU");
      else if (computeMode === "gpu") console.warn("WebGPU 不可用, 回退 CPU 计算(改 ?compute=auto 可静默)");
    }).catch(err => console.warn("WebGPU 初始化异常, 回退 CPU:", err));
  }
  gpuActive = !!(CB && CB.ready && !TORUS && (computeMode === "gpu" || computeMode === "auto"));

  let stepMs = 0;
  if (!paused) {
    const t0 = performance.now();
    stepAcc += speedMul;                       // 时间倍速: 一帧跑多步
    let nSteps = Math.min(Math.floor(stepAcc), 4);
    stepAcc -= nSteps;
    if (gpuActive) {
      if (nSteps > 0) { stepGPUFrame(); stepAcc = 0; }   // GPU 每 rAF 至多一帧(物理已在 GPU, 不留多步债务)
      nSteps = 1;
    } else {
      for (let s = 0; s < nSteps; s++) step();
    }
    stepMs = (performance.now() - t0) / Math.max(nSteps, 1);   // 单步耗时(除以步数)
  }
  perf.stepMs = perf.stepMs * 0.9 + stepMs * 0.1;
  render();

  if (now - lastPanelT > 160) {                // 计分板 ~6Hz 刷新足够
    lastPanelT = now;
    refreshPanel();
  }
}

/* ---- 第二个 <script> 里的 WebGL 后端: 延迟到第一帧再初始化(那时它已解析完) ---- */
function tryInitGL() {
  if (glTried) return;
  glTried = true;
  if (want2D) { console.info("render=2d: 走 Canvas2D 全路径(用于 A/B 对比)"); return; }
  const B = window.GLBackend;
  if (!B) { console.warn("WebGL 后端脚本缺失, 回退 Canvas2D"); return; }
  const ok = B.init(glScene, document.getElementById("glcv"), forceGL);
  glReady = !!ok;
  perf.gl = glReady;
  if (!ok) console.warn("WebGL 不可用, 回退 Canvas2D 全路径");
}

  starling(i) { return { x: P[2*i], y: P[2*i+1], vx: V[2*i], vy: V[2*i+1], alive: !!alive[i], mobber: !!isMobber[i] }; },
  goose(i) { return { x: P2[2*i], y: P2[2*i+1], alive: !!galive[i] }; },
  hunter(i) {
    const h = i < falcons.length ? falcons[i] : eagle;
    return { x: h.x, y: h.y, vx: h.vx, vy: h.vy, energy: h.energy || 0, dashT: h.dashT || 0,
             mob: h.mob || 0, stun: h.stun || 0, lockIdx: h.lockIdx === undefined ? -1 : h.lockIdx,
             score: h.score, dashEats: h.dashEats || 0 };
  },
};

"use strict";
/* ============================================================================
 * WebGPU Compute 后端(第 6 轮, 依据 webgpu-migration-plan.md Phase 1-3)
 * ----------------------------------------------------------------------------
 * 8 个 dispatch 一个 encoder 提交, 同一 compute pass 内顺序可见:
 *   clearGrid(256) -> countCell(每鸟) -> scanCells(单 workgroup 1024, Hillis-Steele
 *   前缀和, 网格桶数须 <= 1024) -> scatter(计数排序, 桶序不确定, 接受) ->
 *   neighborForces(每鸟 3x3 桶扫 + K 寄存器插入排序, 占大头) ->
 *   finalizeCentroid(1 线程) -> reduceSummary(1024, argmax 密度热点) ->
 *   integrate(积分/被捕食/逃逸/围攻/惊飞, 威胁只有 3 个直接展开)。
 * 回读: 双缓冲 staging, mapAsync 取上一帧 —— AI 消费上一帧摘要, 一帧延迟无感。
 * ========================================================================== */
window.ComputeBackend = (function () {
  let device = null, module_ = null, pipe = null, bindGroup = null;
  let buf = null, staging = [null, null], cur = 0;
  let uni = null, OFF = null, readyFlag = false;
  let submittedN = 0, everSubmitted = false;

  const UNI_BYTES = 128;
  // 几何常量按"开阔世界"固化(GPU 路径仅开阔世界; torus 走 CPU 回退)
  const GW = Math.max(1, Math.ceil((W + 2 * PAD_OPEN) / CELL));
  const GH = Math.max(1, Math.ceil((H + 2 * PAD_OPEN) / CELL));
  const CELLS = GW * GH;

  function fnum(v) { return Number.isInteger(v) ? v + ".0" : String(v); }

  function stagingOffsets() {
    if (OFF) return OFF;
    OFF = { summary: 0, pos: 256 };
    OFF.vel = OFF.pos + MAXN * 8;
    OFF.pv = OFF.vel + MAXN * 8;
    OFF.flags = OFF.pv + MAXN * 8;
    OFF.events = OFF.flags + MAXN * 4;
    OFF.total = OFF.events + MAXN * 4;
    return OFF;
  }

  function buildWGSL() {
    return `
struct Threat {
  pos: vec2f,
  vel: vec2f,
  meta: vec2f,          // x = dashT, y = lockIdx (f32, -1 = 无锁定)
  pad: vec2f,
}
struct Uni {
  n: u32,
  k: u32,
  frame: u32,
  pad0: u32,
  mig: vec2f,
  pad1: vec2f,
  threats: array<Threat, 3>,
}
struct Summary {
  cnt: atomic<u32>,
  startled: atomic<u32>,
  eventCursor: atomic<u32>,
  pad0: u32,
  sumX: atomic<i32>,         // 定点质心(坐标 x16, 开阔世界坐标小, i32 足够)
  sumY: atomic<i32>,
  cnt2: atomic<u32>,
  pad1: u32,
  kills: array<atomic<u32>, 3>,
  dashEats: array<atomic<u32>, 3>,
  mob: array<atomic<u32>, 3>,
  hotspot: atomic<u32>,
  centroidOut: vec2f,
  pad2: vec2f,
}
@group(0) @binding(0) var<uniform> U: Uni;
@group(0) @binding(1) var<storage, read_write> posB: array<vec2f>;
@group(0) @binding(2) var<storage, read_write> velB: array<vec2f>;
@group(0) @binding(3) var<storage, read_write> pvB: array<vec2f>;
@group(0) @binding(4) var<storage, read_write> accB: array<vec2f>;
@group(0) @binding(5) var<storage, read_write> flags: array<u32>;   // bit0 活 bit1 围攻者 bit2 惊飞
@group(0) @binding(6) var<storage, read_write> cellCount: array<u32>;
@group(0) @binding(7) var<storage, read_write> cellStart: array<u32>;
@group(0) @binding(8) var<storage, read_write> cellCursor: array<u32>;
@group(0) @binding(9) var<storage, read_write> sorted: array<u32>;
@group(0) @binding(10) var<storage, read_write> events: array<u32>;
@group(0) @binding(11) var<storage, read_write> summary: Summary;

const W: f32 = ${fnum(W)};
const H: f32 = ${fnum(H)};
const PAD: f32 = ${fnum(PAD_OPEN)};
const CELL: f32 = ${fnum(CELL)};
const GWu: u32 = ${GW}u;
const GHu: u32 = ${GH}u;
const GW32: i32 = ${GW};
const GH32: i32 = ${GH};
const CELLSN: u32 = ${CELLS}u;
const MAXNU: u32 = ${MAXN}u;
const PERCEPTION_R2: f32 = ${fnum(PERCEPTION_R2)};
const SEP_R2: f32 = ${fnum(SEP_R2)};
const SEP_GAIN: f32 = ${fnum(SEP_GAIN)};
const MAX_SPEED: f32 = ${fnum(MAX_SPEED)};
const MAX_FORCE: f32 = ${fnum(MAX_FORCE)};
const COHESION_W: f32 = ${fnum(COHESION_W)};
const ALIGNMENT_W: f32 = ${fnum(ALIGNMENT_W)};
const CENTROID_FORCE: f32 = ${fnum(CENTROID_FORCE)};
const BOUND_FORCE: f32 = ${fnum(BOUND_FORCE)};
const EAT_R2_S: f32 = ${fnum(EAT_R2_S)};
const FLEE_R2: f32 = ${fnum(FLEE_R2)};
const FLEE_FORCE: f32 = ${fnum(FLEE_FORCE)};
const FLEE_W: f32 = ${fnum(FLEE_W)};
const MOB_R: f32 = ${fnum(MOB_R)};
const MOB_HARASS: f32 = ${fnum(MOB_HARASS)};
const MOB_MAX: f32 = ${fnum(MOB_MAX)};
const PV_ALERT_COS: f32 = ${fnum(PV_ALERT_COS)};

fn cellOf(p: vec2f) -> u32 {
  let gx = clamp(i32(floor((p.x + PAD) / CELL)), 0, GW32 - 1);
  let gy = clamp(i32(floor((p.y + PAD) / CELL)), 0, GH32 - 1);
  return u32(gy) * GWu + u32(gx);
}

@compute @workgroup_size(256)
fn clearGrid(@builtin(global_invocation_id) gid: vec3u) {
  let t = gid.x;
  for (var c = t; c < CELLSN; c += 256u) {
    cellCount[c] = 0u;
    cellCursor[c] = 0u;
  }
  if (t < 16u) {
    switch t {
      case 0u: { atomicStore(&summary.cnt, 0u); }
      case 1u: { atomicStore(&summary.startled, 0u); }
      case 2u: { atomicStore(&summary.eventCursor, 0u); }
      case 4u: { atomicStore(&summary.sumX, 0); }
      case 5u: { atomicStore(&summary.sumY, 0); }
      case 6u: { atomicStore(&summary.kills[0], 0u); }
      case 7u: { atomicStore(&summary.kills[1], 0u); }
      case 8u: { atomicStore(&summary.kills[2], 0u); }
      case 9u: { atomicStore(&summary.dashEats[0], 0u); }
      case 10u: { atomicStore(&summary.dashEats[1], 0u); }
      case 11u: { atomicStore(&summary.dashEats[2], 0u); }
      case 12u: { atomicStore(&summary.mob[0], 0u); }
      case 13u: { atomicStore(&summary.mob[1], 0u); }
      case 14u: { atomicStore(&summary.mob[2], 0u); }
      case 15u: { atomicStore(&summary.hotspot, 0u); }
      default: {}
    }
  }
}

@compute @workgroup_size(256)
fn countCell(@builtin(global_invocation_id) gid: vec3u) {
  let i = gid.x;
  if (i >= U.n) { return; }
  if ((flags[i] & 1u) == 0u) { return; }
  atomicAdd(&cellCount[cellOf(posB[i])], 1u);
}

var<workgroup> scanBuf: array<u32, 1024>;

@compute @workgroup_size(1024)
fn scanCells(@builtin(local_invocation_id) lid: vec3u) {
  let t = lid.x;
  var v = 0u;
  if (t < CELLSN) { v = cellCount[t]; }
  scanBuf[t] = v;
  workgroupBarrier();
  var offset = 1u;
  loop {
    if (offset >= 1024u) { break; }
    var y = 0u;
    if (t >= offset) { y = scanBuf[t - offset]; }
    workgroupBarrier();
    if (t >= offset) { scanBuf[t] = scanBuf[t] + y; }
    workgroupBarrier();
    offset = offset << 1u;
  }
  if (t < CELLSN) {
    let excl = scanBuf[t] - v;      // 排他前缀和 = 该桶在 sorted 里的起始下标
    cellStart[t] = excl;
    cellCursor[t] = excl;           // scatter 阶段当游标, 扫完后 = 桶尾
  }
}

@compute @workgroup_size(256)
fn scatter(@builtin(global_invocation_id) gid: vec3u) {
  let i = gid.x;
  if (i >= U.n) { return; }
  if ((flags[i] & 1u) == 0u) { return; }
  let c = cellOf(posB[i]);
  let slot = atomicAdd(&cellCursor[c], 1u);
  sorted[slot] = i;
}

@compute @workgroup_size(256)
fn neighborForces(@builtin(global_invocation_id) gid: vec3u) {
  let i = gid.x;
  if (i >= U.n) { return; }
  if ((flags[i] & 1u) == 0u) { return; }
  let p = posB[i];
  let c0 = cellOf(p);
  let cx = i32(c0 % GWu);
  let cy = i32(c0 / GWu);
  var kd: array<f32, 16>;       // TOPO_MAX = 16, K 由 uniform 传入
  var kv: array<vec2f, 16>;
  var kj: array<u32, 16>;
  var kn = 0u;
  for (var oy = -1; oy <= 1; oy++) {
    for (var ox = -1; ox <= 1; ox++) {
      let gx = cx + ox;
      let gy = cy + oy;
      if (gx < 0 || gx >= GW32 || gy < 0 || gy >= GH32) { continue; }
      let cc = u32(gy) * GWu + u32(gx);
      let s0 = cellStart[cc];
      let s1 = cellCursor[cc];
      for (var sIdx = s0; sIdx < s1; sIdx++) {
        let j = sorted[sIdx];
        if (j == i) { continue; }
        let d = posB[j] - p;               // 开阔世界: 无最小像
        let d2 = dot(d, d);
        if (d2 > PERCEPTION_R2) { continue; }
        if (kn < U.k) {                    // 槽未满: 尾部插入 + 冒泡升序
          var tt = kn;
          kn = kn + 1u;
          kd[tt] = d2; kv[tt] = d; kj[tt] = j;
          loop {
            if (tt == 0u || kd[tt - 1u] <= kd[tt]) { break; }
            let td = kd[tt - 1u]; kd[tt - 1u] = kd[tt]; kd[tt] = td;
            let tv = kv[tt - 1u]; kv[tt - 1u] = kv[tt]; kv[tt] = tv;
            let tj = kj[tt - 1u]; kj[tt - 1u] = kj[tt]; kj[tt] = tj;
            tt = tt - 1u;
          }
        } else if (d2 < kd[U.k - 1u]) {    // 比当前第 K 近还近: 顶掉再冒泡
          var tt = U.k - 1u;
          kd[tt] = d2; kv[tt] = d; kj[tt] = j;
          loop {
            if (tt == 0u || kd[tt - 1u] <= kd[tt]) { break; }
            let td = kd[tt - 1u]; kd[tt - 1u] = kd[tt]; kd[tt] = td;
            let tv = kv[tt - 1u]; kv[tt - 1u] = kv[tt]; kv[tt] = tv;
            let tj = kj[tt - 1u]; kj[tt - 1u] = kj[tt]; kj[tt] = tj;
            tt = tt - 1u;
          }
        }
      }
    }
  }
  var a = vec2f(0.0, 0.0);
  if (kn > 0u) {
    var ps = vec2f(0.0, 0.0);
    var vs = vec2f(0.0, 0.0);
    var sep = vec2f(0.0, 0.0);
    for (var m = 0u; m < kn; m++) {
      ps = ps + kv[m];
      vs = vs + velB[kj[m]];
      if (kd[m] < SEP_R2) {
        let w = 1.0 / max(kd[m], 1.0);
        sep = sep - kv[m] * w;
      }
    }
    let desired = vec2f(COHESION_W) * ps / f32(kn) + vec2f(ALIGNMENT_W) * vs / f32(kn);
    let dn = length(desired);
    if (dn > 1e-9) {
      a = a + clamp(desired / dn * MAX_SPEED - velB[i], vec2f(-MAX_FORCE), vec2f(MAX_FORCE));
    }
    a = a + clamp(sep * SEP_GAIN, vec2f(-2.0 * MAX_FORCE), vec2f(2.0 * MAX_FORCE));
  }
  accB[i] = a;
  atomicAdd(&summary.sumX, i32(p.x * 16.0));
  atomicAdd(&summary.sumY, i32(p.y * 16.0));
  atomicAdd(&summary.cnt, 1u);
}

@compute @workgroup_size(1)
fn finalizeCentroid() {
  let c = atomicLoad(&summary.cnt);
  var cxy = vec2f(W * 0.5, H * 0.5);
  if (c > 0u) {
    cxy = vec2f(f32(atomicLoad(&summary.sumX)) / 16.0 / f32(c),
                f32(atomicLoad(&summary.sumY)) / 16.0 / f32(c));
  }
  summary.centroidOut = cxy;
}

var<workgroup> hotCnt: array<u32, 1024>;
var<workgroup> hotIdx: array<u32, 1024>;

@compute @workgroup_size(1024)
fn reduceSummary(@builtin(local_invocation_id) lid: vec3u) {
  let t = lid.x;
  var cnt = 0u;
  if (t < CELLSN) { cnt = cellCount[t]; }
  hotCnt[t] = cnt;
  hotIdx[t] = t;
  workgroupBarrier();
  var stride = 512u;
  loop {
    if (stride == 0u) { break; }
    if (t < stride) {
      if (hotCnt[t + stride] > hotCnt[t]) {
        hotCnt[t] = hotCnt[t + stride];
        hotIdx[t] = hotIdx[t + stride];
      }
    }
    workgroupBarrier();
    stride = stride >> 1u;
  }
  if (t == 0u) { atomicStore(&summary.hotspot, hotIdx[0]); }
}

@compute @workgroup_size(256)
fn integrate(@builtin(global_invocation_id) gid: vec3u) {
  let i = gid.x;
  if (i >= U.n) { return; }
  let fl = flags[i];
  if ((fl & 1u) == 0u) { return; }
  let p = posB[i];
  let v = velB[i];
  var a = accB[i];

  let toC = summary.centroidOut - p;
  let dn = length(toC);
  if (dn > 1e-9) { a = a + toC / dn * CENTROID_FORCE; }
  a = a + U.mig;
  if (p.x < -PAD) { a.x = a.x + BOUND_FORCE; } else if (p.x > W + PAD) { a.x = a.x - BOUND_FORCE; }
  if (p.y < -PAD) { a.y = a.y + BOUND_FORCE; } else if (p.y > H + PAD) { a.y = a.y - BOUND_FORCE; }

  // 被捕食: D 迷惑效应(冲刺锁定单目标), 威胁只有 3 个直接展开
  for (var t = 0u; t < 3u; t++) {
    let th = U.threats[t];
    let dashT = th.meta.x;
    let lockIdx = th.meta.y;
    if (dashT > 0.0 && lockIdx >= 0.0 && u32(lockIdx) != i) { continue; }
    let dd = p - th.pos;
    if (dot(dd, dd) < EAT_R2_S) {
      atomicAdd(&summary.kills[t], 1u);
      if (dashT > 0.0) { atomicAdd(&summary.dashEats[t], 1u); }
      let slot = atomicAdd(&summary.eventCursor, 1u);
      if (slot < MAXNU) { events[slot] = i; }
      flags[i] = fl & 0xFFFFFFFEu;      // 清 alive, 保留围攻者身份位
      return;                            // 死鸟不再逃逸/围攻/积分(CPU 6.4 同: eaten -> continue)
    }
  }
  // 逃逸: 视野内(= 全图判定半径)所有猛禽都生效
  for (var t = 0u; t < 3u; t++) {
    let dd = p - U.threats[t].pos;
    let dp2 = dot(dd, dd);
    if (dp2 < FLEE_R2 && dp2 > 0.0) {
      let dp = sqrt(dp2);
      a = a + FLEE_W * clamp(dd / dp * MAX_SPEED - v, vec2f(-FLEE_FORCE), vec2f(FLEE_FORCE));
    }
  }
  // C 围攻: 围攻者逼近最近猛禽的骚扰环, 环外压/环内推
  if ((fl & 2u) != 0u) {
    var bt = 0u;
    var md2 = 1e18;
    var md = vec2f(0.0, 0.0);
    for (var t = 0u; t < 3u; t++) {
      let dd = p - U.threats[t].pos;
      let d2 = dot(dd, dd);
      if (d2 < md2) { md2 = d2; bt = t; md = dd; }
    }
    let ml = sqrt(md2);
    if (ml < MOB_R && ml > 1e-6) {
      atomicAdd(&summary.mob[bt], 1u);
      let u = md / ml;
      let push = (ml - MOB_HARASS) / MOB_HARASS;
      let tgt = -u * push * MAX_SPEED * 1.5;
      a = a + clamp(tgt - v, vec2f(-MOB_MAX), vec2f(MOB_MAX));
    }
  }
  var nv = v + a;
  let sp = length(nv);
  if (sp > MAX_SPEED) { nv = nv / sp * MAX_SPEED; }
  posB[i] = vec2f(clamp(p.x + nv.x, -PAD - 60.0, W + PAD + 60.0),
                  clamp(p.y + nv.y, -PAD - 60.0, H + PAD + 60.0));
  velB[i] = nv;
  // 惊飞判据: 必须在新速度写入前读旧 pv(顺序与 CPU 6.4 末尾一致)
  let oldv = pvB[i];
  let ops = length(oldv);
  var st = 0u;
  if (ops > 1e-6 && sp > 1e-6 && dot(nv, oldv) / (sp * ops) < PV_ALERT_COS) { st = 1u; }
  if (st == 1u) { atomicAdd(&summary.startled, 1u); }
  flags[i] = (fl & 0xFFFFFFFBu) | (st << 2u);
  pvB[i] = nv;
}
`;
  }

  async function init() {
    if (readyFlag) return true;
    if (!navigator.gpu) return false;
    if (CELLS > 1024) { console.warn("网格桶数 " + CELLS + " > 1024, 单 workgroup 前缀和放不下, 禁用 GPU 路径"); return false; }
    const adapter = await navigator.gpu.requestAdapter({ powerPreference: "high-performance" });
    if (!adapter) return false;
    device = await adapter.requestDevice();
    device.lost.then(info => {
      console.warn("WebGPU 设备丢失, 回退 CPU:", info.message);
      device = null; readyFlag = false;
    });
    module_ = device.createShaderModule({ code: buildWGSL() });
    const entries = [{ binding: 0, visibility: GPUShaderStage.COMPUTE, buffer: { type: "uniform" } }];
    for (let b = 1; b <= 11; b++) entries.push({ binding: b, visibility: GPUShaderStage.COMPUTE, buffer: { type: "storage" } });
    const layout = device.createBindGroupLayout({ entries });
    const pl = device.createPipelineLayout({ bindGroupLayouts: [layout] });
    pipe = {};
    const EPS = ["clearGrid", "countCell", "scanCells", "scatter", "neighborForces", "finalizeCentroid", "reduceSummary", "integrate"];
    await Promise.all(EPS.map(async ep => {
      pipe[ep] = await device.createComputePipelineAsync({ layout: pl, compute: { module: module_, entryPoint: ep } });
    }));
    const S = GPUBufferUsage.STORAGE;
    buf = {
      pos: device.createBuffer({ size: MAXN * 8, usage: S | GPUBufferUsage.COPY_SRC | GPUBufferUsage.COPY_DST }),
      vel: device.createBuffer({ size: MAXN * 8, usage: S | GPUBufferUsage.COPY_SRC | GPUBufferUsage.COPY_DST }),
      pv: device.createBuffer({ size: MAXN * 8, usage: S | GPUBufferUsage.COPY_SRC | GPUBufferUsage.COPY_DST }),
      acc: device.createBuffer({ size: MAXN * 8, usage: S | GPUBufferUsage.COPY_SRC }),
      flags: device.createBuffer({ size: MAXN * 4, usage: S | GPUBufferUsage.COPY_SRC | GPUBufferUsage.COPY_DST }),
      cellCount: device.createBuffer({ size: CELLS * 4, usage: S }),
      cellStart: device.createBuffer({ size: CELLS * 4, usage: S }),
      cellCursor: device.createBuffer({ size: CELLS * 4, usage: S }),
      sorted: device.createBuffer({ size: MAXN * 4, usage: S }),
      events: device.createBuffer({ size: MAXN * 4, usage: S }),
      summary: device.createBuffer({ size: 256, usage: S | GPUBufferUsage.COPY_SRC }),
    };
    uni = device.createBuffer({ size: UNI_BYTES, usage: GPUBufferUsage.UNIFORM | GPUBufferUsage.COPY_DST });
    const off = stagingOffsets();
    for (let k = 0; k < 2; k++)
      staging[k] = device.createBuffer({ size: off.total, usage: GPUBufferUsage.MAP_READ | GPUBufferUsage.COPY_DST });
    const mk = (b) => ({ buffer: buf[b] });
    bindGroup = device.createBindGroup({ layout, entries: [
      { binding: 0, resource: { buffer: uni } },
      { binding: 1, resource: mk("pos") }, { binding: 2, resource: mk("vel") },
      { binding: 3, resource: mk("pv") }, { binding: 4, resource: mk("acc") },
      { binding: 5, resource: mk("flags") }, { binding: 6, resource: mk("cellCount") },
      { binding: 7, resource: mk("cellStart") }, { binding: 8, resource: mk("cellCursor") },
      { binding: 9, resource: mk("sorted") }, { binding: 10, resource: mk("events") },
      { binding: 11, resource: { buffer: buf.summary } },
    ]});
    uploadBirds();
    readyFlag = true;
    return true;
  }

  function uploadBirds() {                   // CPU 侧状态 -> GPU(初始化/重置/补充后调用)
    if (!device) return;
    const n2 = N * 2;
    device.queue.writeBuffer(buf.pos, 0, P.subarray(0, n2));
    device.queue.writeBuffer(buf.vel, 0, V.subarray(0, n2));
    device.queue.writeBuffer(buf.pv, 0, PV.subarray(0, n2));
    const fl = new Uint32Array(N);
    for (let i = 0; i < N; i++) fl[i] = (alive[i] ? 1 : 0) | (isMobber[i] ? 2 : 0) | (startled[i] ? 4 : 0);
    device.queue.writeBuffer(buf.flags, 0, fl);
  }

  function submit() {                        // 一帧物理: 8 个 dispatch + staging 拷贝, 一次提交
    if (!device) return;
    const u = new ArrayBuffer(UNI_BYTES);
    const U32 = new Uint32Array(u), F = new Float32Array(u);
    U32[0] = N; U32[1] = topoK; U32[2] = frame >>> 0;
    F[4] = migX; F[5] = migY;
    for (let t = 0; t < 3; t++) {
      const f = threats[t], o = 8 + t * 8;
      F[o] = f.x; F[o + 1] = f.y; F[o + 2] = f.vx; F[o + 3] = f.vy;
      F[o + 4] = f.dashT; F[o + 5] = f.lockIdx;
    }
    device.queue.writeBuffer(uni, 0, u);
    const enc = device.createCommandEncoder();
    const pass = enc.beginComputePass();
    pass.setBindGroup(0, bindGroup);
    const wg = Math.ceil(N / 256);
    pass.setPipeline(pipe.clearGrid);        pass.dispatchWorkgroups(1);
    pass.setPipeline(pipe.countCell);        pass.dispatchWorkgroups(wg);
    pass.setPipeline(pipe.scanCells);        pass.dispatchWorkgroups(1);
    pass.setPipeline(pipe.scatter);          pass.dispatchWorkgroups(wg);
    pass.setPipeline(pipe.neighborForces);   pass.dispatchWorkgroups(wg);
    pass.setPipeline(pipe.finalizeCentroid); pass.dispatchWorkgroups(1);
    pass.setPipeline(pipe.reduceSummary);    pass.dispatchWorkgroups(1);
    pass.setPipeline(pipe.integrate);        pass.dispatchWorkgroups(wg);
    pass.end();
    const off = stagingOffsets(), st = staging[cur];
    const n2b = N * 8, nb = N * 4;
    enc.copyBufferToBuffer(buf.pos, 0, st, off.pos, n2b);
    enc.copyBufferToBuffer(buf.vel, 0, st, off.vel, n2b);
    enc.copyBufferToBuffer(buf.pv, 0, st, off.pv, n2b);
    enc.copyBufferToBuffer(buf.flags, 0, st, off.flags, nb);
    enc.copyBufferToBuffer(buf.events, 0, st, off.events, nb);
    enc.copyBufferToBuffer(buf.summary, 0, st, 0, 256);
    device.queue.submit([enc.finish()]);
    submittedN = N; everSubmitted = true;
    cur = 1 - cur;
  }

  async function readbackPrev() {            // 取上一帧提交的回读(mapAsync 等待完成)
    if (!device || !everSubmitted) return null;
    const other = 1 - cur;
    const st = staging[other];
    await st.mapAsync(GPUMapMode.READ);
    const ab = st.getMappedRange();
    const off = stagingOffsets();
    const n = submittedN;
    P.set(new Float32Array(ab, off.pos, n * 2));
    V.set(new Float32Array(ab, off.vel, n * 2));
    PV.set(new Float32Array(ab, off.pv, n * 2));
    const FL = new Uint32Array(ab, off.flags, n);
    const cnt = new Uint32Array(ab, 0, 16);
    let sx = 0, sy = 0, c = 0;
    for (let i = 0; i < n; i++) {
      const f = FL[i];
      alive[i] = f & 1;
      isMobber[i] = (f >> 1) & 1;
      startled[i] = (f >> 2) & 1;
      if (f & 1) { sx += P[2*i]; sy += P[2*i+1]; c++; }
    }
    const meta = {
      n, cnt: cnt[0],
      cx: c ? sx / c : W / 2, cy: c ? sy / c : H / 2,
      events: Math.min(cnt[2], n),
      evIdx: new Uint32Array(ab, off.events, n).slice(0, Math.min(cnt[2], n)),
      kills: [cnt[6], cnt[7], cnt[8]],
      dashEats: [cnt[9], cnt[10], cnt[11]],
      mob: [cnt[12], cnt[13], cnt[14]],
      hotspot: cnt[15],
    };
    st.unmap();
    return meta;
  }

  return {
    init, submit, readbackPrev, uploadBirds,
    get ready() { return readyFlag; },
    info() { return device ? { cells: CELLS, stagingBytes: stagingOffsets().total } : null; },
  };
})();

 ============================ 第 4 轮之一: 采样目标获取 ============================
 问题: 最密网格桶有量化误差; 质心可能正好落在隼自己犁出的空腔里; 也无法扩展风险项。
 做法(P1): perceive() 在收集可见椋鸟的同时, 以猛禽为原点布 5 环 × 16 角 = 80 个极坐标
   候选点, 每只鸟累加进"距离 < SCOOP_R 的候选点", 得到一个 80 维的"捞勺计数"。
   打分公式:
     score(c) = (c 的捞勺内鸟数) × (1 + DIR_LAG·cos(候选方向, 当前航向))
                × [锥内乘项]        // 见第 4 轮之二 D
                // 扩展位: 雁权重 / 威慑锥扣分 / mob 扣分
   方向滞后项让目标不会在两个等分候选点之间抖动; 锥内乘项让"选目标"自己绕开雁群正面,
   而不是写一条"猛禽看见雁就躲"的硬编码。
   取 argmax 候选点为冲刺中心 (cx,cy); LOCK_R=40px 内最近的鸟为 lockIdx,
   40px 内没有鸟就放弃这次冲刺(不做无效冲刺)。tryDash 的 lockIdx 语义不变:
   冲刺途中只吃得到被锁定的那一只(雁的 EAT 判定不加, 雁大不存在迷惑);
   互攻/驱逐冲刺传 -1, 冲刺结束 lockIdx 归 -1。
   常量: SAMPLE_RADII={80,160,240,320,400} SAMPLE_ANGLES=16 SCOOP_R=60 LOCK_R=40
         DIR_LAG=0.3  (候选距猛禽 > VIEW_R - SCOOP_R 的直接跳过)
   性能: 内层用"半径平方 + 预计算角度窗(CAND_W)"把每只鸟的候选测试从 80 次压到 5~10 次,
   N=2000 下整步约 9~11ms。

 ============================ 第 4 轮之二: 雁群中心化重构 ============================
 目标: 雁从"有槽位的椋鸟"变成真正的中心化组织 —— 响应即时、无单点故障、有威慑,
   但保留制度性弱点。
 A 领队智能: LEADER_SENSE_R=400 内最近猛禽 -> leaderAng 以 LEADER_MAX_TURN=0.04 rad/帧
   限速转向远离; 无威胁时随机游动 + 向心; 威胁 < LEADER_PANIC_R=250 时全队速度
   ×ESCAPE_HUSTLE=1.15。每只雁另有 GOOSE_MAX_TURN=0.06 rad/帧 的转向限速(压坡惯性)。
 B 轮值领队: ROTATE_FRAMES=1200(20 秒)换一次 apex; 槽位号 slot(i) = (i + gooseShift) % M,
   虚拟领队位置就是当前 apex 雁的位置(所以换岗不瞬移, leaderAng 继承)。没有可斩首的固定领队。
   幼雁内道靠"残类"实现: 幼雁是固定个体(i % 4 === 0, 25%), 而 4 | M=60, 所以它们的槽位
   残类恒等于 gooseShift % 4 —— 把这一残类的槽位排成内道(小排号), 幼雁就永远在内道。
 C 核心保护: JUVENILE_FRAC=0.25; 幼雁死亡 -> JUVENILE_DEBUFF_FRAMES=1800 帧内全体归位力
   ×DEBUFF_FACTOR=0.7(沮丧期)。
 D 威慑锥: 顶点 = V 尖, 轴 = 领队航向, 半角 60°, 半径 CONE_R=180。两个效果:
   a) 锥内候选点打分 ×CONE_SCORE_FACTOR=0.25(与之一咬合, 选目标自己绕开正面);
   b) 锥内猛禽持续失血: 隼冲刺能耗 ×CONE_FALCON_DASH_MULT=2、鹰每帧 pending -= 0.01

 E 斩首混乱: 当前 apex 被吃的瞬间 disarray = DISARRAY_FRAMES=90: 槽位力关闭, 全体退化为

 ============================ 第 4 轮之三: 隼的策略进化 ============================
 原则: 制度(能量轮换/拦截位/交棒/驱逐)不学, 学的是制度内的操作参数 —— 鹰学价值(Q-learning),
   隼学策略(进化), 两种学习哲学同框。
 可学习参数(每只隼一份, 归一化 [0,1] 存储):
   p1 出手能量阈值 -> DASH_MIN     ∈[20,60] 默认 35
   p2 交棒能量阈值 -> ENERGY_LOW   ∈[10,40] 默认 25
   p3 双打能量阈值 -> DOUBLE_TEAM_E∈[60,95] 默认 85
   p4 拦截位外推   -> INTERCEPT_AHEAD ∈[0.3,0.8] 默认 0.55

   p6 线性冲刺策略(4 维特征 + 阈值)留作扩展位, 未实现 —— 若要做, 在 6.8 的出手判定处
      把 PV_.dashMin 比较换成 w·[能量, 目标密度, 1-mob, 航向对齐度] > θ, 并把 w/θ 塞进 ES 向量。
 算法: (1+1)-ES 在线进化, ES_EVAL_FRAMES=3600(60 秒)一个评估期(适应度噪声大, 期长配对比较)。
   双缓冲: 偶数期评估当前参数, 奇数期评估"当前 + N(0, ES_SIGMA=0.1) 截断到 [0,1]"的候选;

        1, 0, -1, -1.28, -1, -0.72,     // 右臂: 鼻尖 -> 右翼外缘 -> 右翼内缘

      maxN = s.maxN || 20000; maxM = s.maxM || 60;
      starBuf = new Float32Array(maxN * 3);
      gooseBuf = new Float32Array(maxM * 3);
      instVBO = gl.createBuffer();
      gl.bindBuffer(gl.ARRAY_BUFFER, instVBO);
      gl.bufferData(gl.ARRAY_BUFFER, starBuf.byteLength, gl.DYNAMIC_DRAW);   // 按最大 N 预留

      texA = makeTex(s.W, s.H); texB = makeTex(s.W, s.H);
      fboA = makeFBO(texA); fboB = makeFBO(texB);
      if (!fboA || !fboB) { console.warn("离屏帧缓冲不完整, 回退 Canvas2D"); return false; }

      for (let k = 0; k < 2; k++) {
        gl.bindFramebuffer(gl.FRAMEBUFFER, k ? fboB : fboA);
        gl.viewport(0, 0, s.W, s.H);
        gl.clearColor(s.bg[0], s.bg[1], s.bg[2], 1);
        gl.clear(gl.COLOR_BUFFER_BIT);

      gl.disable(gl.DEPTH_TEST);
      gl.disable(gl.BLEND);                       // 鸟是不透明色块

      U.shape = gl.getUniformLocation(progBird, "uShape");
      U.color = gl.getUniformLocation(progBird, "uColor");
      U.tex = gl.getUniformLocation(progFade, "uTex");
      U.bg = gl.getUniformLocation(progFade, "uBg");
      U.fade = gl.getUniformLocation(progFade, "uFade");
      U.copy = gl.getUniformLocation(progFade, "uCopy");

      A.inst = gl.getAttribLocation(progBird, "aInst");

      gl.bindBuffer(gl.ARRAY_BUFFER, birdVBO);
      gl.enableVertexAttribArray(A.unit);

      gl.bindBuffer(gl.ARRAY_BUFFER, quadVBO);

    },

      const curFBO = curIsA ? fboA : fboB;

      gl.uniform1f(U.copy, 0);                             // 0 = 衰减模式
      gl.bindBuffer(gl.ARRAY_BUFFER, quadVBO);
      gl.vertexAttribPointer(A.quad, 2, gl.FLOAT, false, 0, 0);
      gl.drawArrays(gl.TRIANGLES, 0, 6);

      /* ---- ② 组装实例缓冲(只装活着的鸟, 3 float/只) ---- */
      const alive = s.alive, P = s.P, V = s.V;
      let n = 0;
      const limN = Math.min(c.n, maxN);
      for (let i = 0; i < limN; i++) {
        if (!alive[i]) continue;
        starBuf[n * 3] = P[2 * i]; starBuf[n * 3 + 1] = P[2 * i + 1];
        starBuf[n * 3 + 2] = Math.atan2(V[2 * i + 1], V[2 * i]);
        n++;
      }
      const gal = s.galive, P2 = s.P2, V2 = s.V2;
      let m = 0;
      const limM = Math.min(c.m, maxM);
      for (let i = 0; i < limM; i++) {
        if (!gal[i]) continue;
        gooseBuf[m * 3] = P2[2 * i]; gooseBuf[m * 3 + 1] = P2[2 * i + 1];
        gooseBuf[m * 3 + 2] = Math.atan2(V2[2 * i + 1], V2[2 * i]);
        m++;
      }

      /* ---- ③ 两次实例化绘制(旋转/缩放/投影都在着色器) ---- */
      gl.useProgram(progBird);
      gl.uniform2f(U.res, s.W, s.H);
      gl.bindBuffer(gl.ARRAY_BUFFER, instVBO);
      gl.vertexAttribPointer(A.inst, 3, gl.FLOAT, false, 0, 0);
      if (n > 0) {
        if (isGL2) gl.bufferSubData(gl.ARRAY_BUFFER, 0, starBuf, 0, n * 3);
        else gl.bufferSubData(gl.ARRAY_BUFFER, 0, starBuf.subarray(0, n * 3));
        gl.uniform3f(U.shape, s.shapes.starling[0], s.shapes.starling[1], s.shapes.starling[2]);
        gl.uniform3f(U.color, s.colors.starling[0], s.colors.starling[1], s.colors.starling[2]);
        if (isGL2) gl.drawArraysInstanced(gl.TRIANGLES, 0, 6, n);
        else angleExt.drawArraysInstancedANGLE(gl.TRIANGLES, 0, 6, n);
      }
      if (m > 0) {
        if (isGL2) gl.bufferSubData(gl.ARRAY_BUFFER, 0, gooseBuf, 0, m * 3);
        else gl.bufferSubData(gl.ARRAY_BUFFER, 0, gooseBuf.subarray(0, m * 3));
        gl.uniform3f(U.shape, s.shapes.goose[0], s.shapes.goose[1], s.shapes.goose[2]);
        gl.uniform3f(U.color, s.colors.goose[0], s.colors.goose[1], s.colors.goose[2]);
        if (isGL2) gl.drawArraysInstanced(gl.TRIANGLES, 0, 6, m);
        else angleExt.drawArraysInstancedANGLE(gl.TRIANGLES, 0, 6, m);
      }

      /* ---- ④ 拷到默认帧缓冲(2D 叠加层覆盖在它上面) ---- */
      gl.bindFramebuffer(gl.FRAMEBUFFER, null);            // 渲染目标 = 屏幕
      gl.useProgram(progFade);
      gl.activeTexture(gl.TEXTURE0);
      gl.bindTexture(gl.TEXTURE_2D, curIsA ? texA : texB); // 本帧刚画好的纹理
      gl.uniform1i(U.tex, 0);
      gl.uniform1f(U.copy, 1);                             // 1 = 原样拷贝
      gl.bindBuffer(gl.ARRAY_BUFFER, quadVBO);
      gl.vertexAttribPointer(A.quad, 2, gl.FLOAT, false, 0, 0);
      gl.drawArrays(gl.TRIANGLES, 0, 6);

      curIsA = !curIsA;                                    // ping-pong 交换
    },
  };
})();

/* ============================================================================
 * WebGPU Compute 后端(第 6 轮, 依据 webgpu-migration-plan.md Phase 1-3)
 * ----------------------------------------------------------------------------
 * 8 个 dispatch 一个 encoder 提交, 同一 compute pass 内顺序可见:
 *   clearGrid(256) -> countCell(每鸟) -> scanCells(单 workgroup 1024, Hillis-Steele
 *   前缀和, 网格桶数须 <= 1024) -> scatter(计数排序, 桶序不确定, 接受) ->
 *   neighborForces(每鸟 3x3 桶扫 + K 寄存器插入排序, 占大头) ->
 *   finalizeCentroid(1 线程) -> reduceSummary(1024, argmax 密度热点) ->
 *   integrate(积分/被捕食/逃逸/围攻/惊飞, 威胁只有 3 个直接展开)。
 * 回读: 双缓冲 staging, mapAsync 取上一帧 —— AI 消费上一帧摘要, 一帧延迟无感。
 * ========================================================================== */
window.ComputeBackend = (function () {
  let device = null, module_ = null, pipe = null, bindGroup = null;
  let buf = null, staging = [null, null], cur = 0;
  let uni = null, OFF = null, readyFlag = false;
  let submittedN = 0, everSubmitted = false;

  const UNI_BYTES = 128;
  // 几何常量按"开阔世界"固化(GPU 路径仅开阔世界; torus 走 CPU 回退)
  const GW = Math.max(1, Math.ceil((W + 2 * PAD_OPEN) / CELL));
  const GH = Math.max(1, Math.ceil((H + 2 * PAD_OPEN) / CELL));
  const CELLS = GW * GH;

  function fnum(v) { return Number.isInteger(v) ? v + ".0" : String(v); }

  function stagingOffsets() {
    if (OFF) return OFF;
    OFF = { summary: 0, pos: 256 };
    OFF.vel = OFF.pos + MAXN * 8;
    OFF.pv = OFF.vel + MAXN * 8;
    OFF.flags = OFF.pv + MAXN * 8;
    OFF.events = OFF.flags + MAXN * 4;
    OFF.total = OFF.events + MAXN * 4;
    return OFF;
  }

  function buildWGSL() {
    return `
struct Threat {
  pos: vec2f,
  vel: vec2f,
  meta: vec2f,          // x = dashT, y = lockIdx (f32, -1 = 无锁定)
  pad: vec2f,
}
struct Uni {
  n: u32,
  k: u32,
  frame: u32,
  pad0: u32,
  mig: vec2f,
  pad1: vec2f,
  threats: array<Threat, 3>,
}
struct Summary {
  cnt: atomic<u32>,
  startled: atomic<u32>,
  eventCursor: atomic<u32>,
  pad0: u32,
  sumX: atomic<i32>,         // 定点质心(坐标 x16, 开阔世界坐标小, i32 足够)
  sumY: atomic<i32>,
  cnt2: atomic<u32>,
  pad1: u32,
  kills: array<atomic<u32>, 3>,
  dashEats: array<atomic<u32>, 3>,
  mob: array<atomic<u32>, 3>,
  hotspot: atomic<u32>,
  centroidOut: vec2f,
  pad2: vec2f,
}
@group(0) @binding(0) var<uniform> U: Uni;
@group(0) @binding(1) var<storage, read_write> posB: array<vec2f>;
@group(0) @binding(2) var<storage, read_write> velB: array<vec2f>;
@group(0) @binding(3) var<storage, read_write> pvB: array<vec2f>;
@group(0) @binding(4) var<storage, read_write> accB: array<vec2f>;
@group(0) @binding(5) var<storage, read_write> flags: array<u32>;   // bit0 活 bit1 围攻者 bit2 惊飞
@group(0) @binding(6) var<storage, read_write> cellCount: array<u32>;
@group(0) @binding(7) var<storage, read_write> cellStart: array<u32>;
@group(0) @binding(8) var<storage, read_write> cellCursor: array<u32>;
@group(0) @binding(9) var<storage, read_write> sorted: array<u32>;
@group(0) @binding(10) var<storage, read_write> events: array<u32>;
@group(0) @binding(11) var<storage, read_write> summary: Summary;

const W: f32 = ${fnum(W)};
const H: f32 = ${fnum(H)};
const PAD: f32 = ${fnum(PAD_OPEN)};
const CELL: f32 = ${fnum(CELL)};
const GWu: u32 = ${GW}u;
const GHu: u32 = ${GH}u;
const GW32: i32 = ${GW};
const GH32: i32 = ${GH};
const CELLSN: u32 = ${CELLS}u;
const MAXNU: u32 = ${MAXN}u;
const PERCEPTION_R2: f32 = ${fnum(PERCEPTION_R2)};
const SEP_R2: f32 = ${fnum(SEP_R2)};
const SEP_GAIN: f32 = ${fnum(SEP_GAIN)};
const MAX_SPEED: f32 = ${fnum(MAX_SPEED)};
const MAX_FORCE: f32 = ${fnum(MAX_FORCE)};
const COHESION_W: f32 = ${fnum(COHESION_W)};
const ALIGNMENT_W: f32 = ${fnum(ALIGNMENT_W)};
const CENTROID_FORCE: f32 = ${fnum(CENTROID_FORCE)};
const BOUND_FORCE: f32 = ${fnum(BOUND_FORCE)};
const EAT_R2_S: f32 = ${fnum(EAT_R2_S)};
const FLEE_R2: f32 = ${fnum(FLEE_R2)};
const FLEE_FORCE: f32 = ${fnum(FLEE_FORCE)};
const FLEE_W: f32 = ${fnum(FLEE_W)};
const MOB_R: f32 = ${fnum(MOB_R)};
const MOB_HARASS: f32 = ${fnum(MOB_HARASS)};
const MOB_MAX: f32 = ${fnum(MOB_MAX)};
const PV_ALERT_COS: f32 = ${fnum(PV_ALERT_COS)};

fn cellOf(p: vec2f) -> u32 {
  let gx = clamp(i32(floor((p.x + PAD) / CELL)), 0, GW32 - 1);
  let gy = clamp(i32(floor((p.y + PAD) / CELL)), 0, GH32 - 1);
  return u32(gy) * GWu + u32(gx);
}

@compute @workgroup_size(256)
fn clearGrid(@builtin(global_invocation_id) gid: vec3u) {
  let t = gid.x;
  for (var c = t; c < CELLSN; c += 256u) {
    cellCount[c] = 0u;
    cellCursor[c] = 0u;
  }
  if (t < 16u) {
    switch t {
      case 0u: { atomicStore(&summary.cnt, 0u); }
      case 1u: { atomicStore(&summary.startled, 0u); }
      case 2u: { atomicStore(&summary.eventCursor, 0u); }
      case 4u: { atomicStore(&summary.sumX, 0); }
      case 5u: { atomicStore(&summary.sumY, 0); }
      case 6u: { atomicStore(&summary.kills[0], 0u); }
      case 7u: { atomicStore(&summary.kills[1], 0u); }
      case 8u: { atomicStore(&summary.kills[2], 0u); }
      case 9u: { atomicStore(&summary.dashEats[0], 0u); }
      case 10u: { atomicStore(&summary.dashEats[1], 0u); }
      case 11u: { atomicStore(&summary.dashEats[2], 0u); }
      case 12u: { atomicStore(&summary.mob[0], 0u); }
      case 13u: { atomicStore(&summary.mob[1], 0u); }
      case 14u: { atomicStore(&summary.mob[2], 0u); }
      case 15u: { atomicStore(&summary.hotspot, 0u); }
      default: {}
    }
  }
}

@compute @workgroup_size(256)
fn countCell(@builtin(global_invocation_id) gid: vec3u) {
  let i = gid.x;
  if (i >= U.n) { return; }
  if ((flags[i] & 1u) == 0u) { return; }
  atomicAdd(&cellCount[cellOf(posB[i])], 1u);
}

var<workgroup> scanBuf: array<u32, 1024>;

@compute @workgroup_size(1024)
fn scanCells(@builtin(local_invocation_id) lid: vec3u) {
  let t = lid.x;
  var v = 0u;
  if (t < CELLSN) { v = cellCount[t]; }
  scanBuf[t] = v;
  workgroupBarrier();
  var offset = 1u;
  loop {
    if (offset >= 1024u) { break; }
    var y = 0u;
    if (t >= offset) { y = scanBuf[t - offset]; }
    workgroupBarrier();
    if (t >= offset) { scanBuf[t] = scanBuf[t] + y; }
    workgroupBarrier();
    offset = offset << 1u;
  }
  if (t < CELLSN) {
    let excl = scanBuf[t] - v;      // 排他前缀和 = 该桶在 sorted 里的起始下标
    cellStart[t] = excl;
    cellCursor[t] = excl;           // scatter 阶段当游标, 扫完后 = 桶尾
  }
}

@compute @workgroup_size(256)
fn scatter(@builtin(global_invocation_id) gid: vec3u) {
  let i = gid.x;
  if (i >= U.n) { return; }
  if ((flags[i] & 1u) == 0u) { return; }
  let c = cellOf(posB[i]);
  let slot = atomicAdd(&cellCursor[c], 1u);
  sorted[slot] = i;
}

@compute @workgroup_size(256)
fn neighborForces(@builtin(global_invocation_id) gid: vec3u) {
  let i = gid.x;
  if (i >= U.n) { return; }
  if ((flags[i] & 1u) == 0u) { return; }
  let p = posB[i];
  let c0 = cellOf(p);
  let cx = i32(c0 % GWu);
  let cy = i32(c0 / GWu);
  var kd: array<f32, 16>;       // TOPO_MAX = 16, K 由 uniform 传入
  var kv: array<vec2f, 16>;
  var kj: array<u32, 16>;
  var kn = 0u;
  for (var oy = -1; oy <= 1; oy++) {
    for (var ox = -1; ox <= 1; ox++) {
      let gx = cx + ox;
      let gy = cy + oy;
      if (gx < 0 || gx >= GW32 || gy < 0 || gy >= GH32) { continue; }
      let cc = u32(gy) * GWu + u32(gx);
      let s0 = cellStart[cc];
      let s1 = cellCursor[cc];
      for (var sIdx = s0; sIdx < s1; sIdx++) {
        let j = sorted[sIdx];
        if (j == i) { continue; }
        let d = posB[j] - p;               // 开阔世界: 无最小像
        let d2 = dot(d, d);
        if (d2 > PERCEPTION_R2) { continue; }
        if (kn < U.k) {                    // 槽未满: 尾部插入 + 冒泡升序
          var tt = kn;
          kn = kn + 1u;
          kd[tt] = d2; kv[tt] = d; kj[tt] = j;
          loop {
            if (tt == 0u || kd[tt - 1u] <= kd[tt]) { break; }
            let td = kd[tt - 1u]; kd[tt - 1u] = kd[tt]; kd[tt] = td;
            let tv = kv[tt - 1u]; kv[tt - 1u] = kv[tt]; kv[tt] = tv;
            let tj = kj[tt - 1u]; kj[tt - 1u] = kj[tt]; kj[tt] = tj;
            tt = tt - 1u;
          }
        } else if (d2 < kd[U.k - 1u]) {    // 比当前第 K 近还近: 顶掉再冒泡
          var tt = U.k - 1u;
          kd[tt] = d2; kv[tt] = d; kj[tt] = j;
          loop {
            if (tt == 0u || kd[tt - 1u] <= kd[tt]) { break; }
            let td = kd[tt - 1u]; kd[tt - 1u] = kd[tt]; kd[tt] = td;
            let tv = kv[tt - 1u]; kv[tt - 1u] = kv[tt]; kv[tt] = tv;
            let tj = kj[tt - 1u]; kj[tt - 1u] = kj[tt]; kj[tt] = tj;
            tt = tt - 1u;
          }
        }
      }
    }
  }
  var a = vec2f(0.0, 0.0);
  if (kn > 0u) {
    var ps = vec2f(0.0, 0.0);
    var vs = vec2f(0.0, 0.0);
    var sep = vec2f(0.0, 0.0);
    for (var m = 0u; m < kn; m++) {
      ps = ps + kv[m];
      vs = vs + velB[kj[m]];
      if (kd[m] < SEP_R2) {
        let w = 1.0 / max(kd[m], 1.0);
        sep = sep - kv[m] * w;
      }
    }
    let desired = vec2f(COHESION_W) * ps / f32(kn) + vec2f(ALIGNMENT_W) * vs / f32(kn);
    let dn = length(desired);
    if (dn > 1e-9) {
      a = a + clamp(desired / dn * MAX_SPEED - velB[i], vec2f(-MAX_FORCE), vec2f(MAX_FORCE));
    }
    a = a + clamp(sep * SEP_GAIN, vec2f(-2.0 * MAX_FORCE), vec2f(2.0 * MAX_FORCE));
  }
  accB[i] = a;
  atomicAdd(&summary.sumX, i32(p.x * 16.0));
  atomicAdd(&summary.sumY, i32(p.y * 16.0));
  atomicAdd(&summary.cnt, 1u);
}

@compute @workgroup_size(1)
fn finalizeCentroid() {
  let c = atomicLoad(&summary.cnt);
  var cxy = vec2f(W * 0.5, H * 0.5);
  if (c > 0u) {
    cxy = vec2f(f32(atomicLoad(&summary.sumX)) / 16.0 / f32(c),
                f32(atomicLoad(&summary.sumY)) / 16.0 / f32(c));
  }
  summary.centroidOut = cxy;
}

var<workgroup> hotCnt: array<u32, 1024>;
var<workgroup> hotIdx: array<u32, 1024>;

@compute @workgroup_size(1024)
fn reduceSummary(@builtin(local_invocation_id) lid: vec3u) {
  let t = lid.x;
  var cnt = 0u;
  if (t < CELLSN) { cnt = cellCount[t]; }
  hotCnt[t] = cnt;
  hotIdx[t] = t;
  workgroupBarrier();
  var stride = 512u;
  loop {
    if (stride == 0u) { break; }
    if (t < stride) {
      if (hotCnt[t + stride] > hotCnt[t]) {
        hotCnt[t] = hotCnt[t + stride];
        hotIdx[t] = hotIdx[t + stride];
      }
    }
    workgroupBarrier();
    stride = stride >> 1u;
  }
  if (t == 0u) { atomicStore(&summary.hotspot, hotIdx[0]); }
}

@compute @workgroup_size(256)
fn integrate(@builtin(global_invocation_id) gid: vec3u) {
  let i = gid.x;
  if (i >= U.n) { return; }
  let fl = flags[i];
  if ((fl & 1u) == 0u) { return; }
  let p = posB[i];
  let v = velB[i];
  var a = accB[i];

  let toC = summary.centroidOut - p;
  let dn = length(toC);
  if (dn > 1e-9) { a = a + toC / dn * CENTROID_FORCE; }
  a = a + U.mig;
  if (p.x < -PAD) { a.x = a.x + BOUND_FORCE; } else if (p.x > W + PAD) { a.x = a.x - BOUND_FORCE; }
  if (p.y < -PAD) { a.y = a.y + BOUND_FORCE; } else if (p.y > H + PAD) { a.y = a.y - BOUND_FORCE; }

  // 被捕食: D 迷惑效应(冲刺锁定单目标), 威胁只有 3 个直接展开
  for (var t = 0u; t < 3u; t++) {
    let th = U.threats[t];
    let dashT = th.meta.x;
    let lockIdx = th.meta.y;
    if (dashT > 0.0 && lockIdx >= 0.0 && u32(lockIdx) != i) { continue; }
    let dd = p - th.pos;
    if (dot(dd, dd) < EAT_R2_S) {
      atomicAdd(&summary.kills[t], 1u);
      if (dashT > 0.0) { atomicAdd(&summary.dashEats[t], 1u); }
      let slot = atomicAdd(&summary.eventCursor, 1u);
      if (slot < MAXNU) { events[slot] = i; }
      flags[i] = fl & 0xFFFFFFFEu;      // 清 alive, 保留围攻者身份位
      return;                            // 死鸟不再逃逸/围攻/积分(CPU 6.4 同: eaten -> continue)
    }
  }
  // 逃逸: 视野内(= 全图判定半径)所有猛禽都生效
  for (var t = 0u; t < 3u; t++) {
    let dd = p - U.threats[t].pos;
    let dp2 = dot(dd, dd);
    if (dp2 < FLEE_R2 && dp2 > 0.0) {
      let dp = sqrt(dp2);
      a = a + FLEE_W * clamp(dd / dp * MAX_SPEED - v, vec2f(-FLEE_FORCE), vec2f(FLEE_FORCE));
    }
  }
  // C 围攻: 围攻者逼近最近猛禽的骚扰环, 环外压/环内推
  if ((fl & 2u) != 0u) {
    var bt = 0u;
    var md2 = 1e18;
    var md = vec2f(0.0, 0.0);
    for (var t = 0u; t < 3u; t++) {
      let dd = p - U.threats[t].pos;
      let d2 = dot(dd, dd);
      if (d2 < md2) { md2 = d2; bt = t; md = dd; }
    }
    let ml = sqrt(md2);
    if (ml < MOB_R && ml > 1e-6) {
      atomicAdd(&summary.mob[bt], 1u);
      let u = md / ml;
      let push = (ml - MOB_HARASS) / MOB_HARASS;
      let tgt = -u * push * MAX_SPEED * 1.5;
      a = a + clamp(tgt - v, vec2f(-MOB_MAX), vec2f(MOB_MAX));
    }
  }
  var nv = v + a;
  let sp = length(nv);
  if (sp > MAX_SPEED) { nv = nv / sp * MAX_SPEED; }
  posB[i] = vec2f(clamp(p.x + nv.x, -PAD - 60.0, W + PAD + 60.0),
                  clamp(p.y + nv.y, -PAD - 60.0, H + PAD + 60.0));
  velB[i] = nv;
  // 惊飞判据: 必须在新速度写入前读旧 pv(顺序与 CPU 6.4 末尾一致)
  let oldv = pvB[i];
  let ops = length(oldv);
  var st = 0u;
  if (ops > 1e-6 && sp > 1e-6 && dot(nv, oldv) / (sp * ops) < PV_ALERT_COS) { st = 1u; }
  if (st == 1u) { atomicAdd(&summary.startled, 1u); }
  flags[i] = (fl & 0xFFFFFFFBu) | (st << 2u);
  pvB[i] = nv;
}
`;

  async function init() {
    if (readyFlag) return true;
    if (!navigator.gpu) return false;
    if (CELLS > 1024) { console.warn("网格桶数 " + CELLS + " > 1024, 单 workgroup 前缀和放不下, 禁用 GPU 路径"); return false; }
    const adapter = await navigator.gpu.requestAdapter({ powerPreference: "high-performance" });
    if (!adapter) return false;
    device = await adapter.requestDevice();
    device.lost.then(info => {
      console.warn("WebGPU 设备丢失, 回退 CPU:", info.message);
      device = null; readyFlag = false;
    });
    module_ = device.createShaderModule({ code: buildWGSL() });
    const entries = [{ binding: 0, visibility: GPUShaderStage.COMPUTE, buffer: { type: "uniform" } }];
    for (let b = 1; b <= 11; b++) entries.push({ binding: b, visibility: GPUShaderStage.COMPUTE, buffer: { type: "storage" } });
    const layout = device.createBindGroupLayout({ entries });
    const pl = device.createPipelineLayout({ bindGroupLayouts: [layout] });
    pipe = {};
    const EPS = ["clearGrid", "countCell", "scanCells", "scatter", "neighborForces", "finalizeCentroid", "reduceSummary", "integrate"];
    await Promise.all(EPS.map(async ep => {
      pipe[ep] = await device.createComputePipelineAsync({ layout: pl, compute: { module: module_, entryPoint: ep } });
    }));
    const S = GPUBufferUsage.STORAGE;
    buf = {
      pos: device.createBuffer({ size: MAXN * 8, usage: S | GPUBufferUsage.COPY_SRC | GPUBufferUsage.COPY_DST }),
      vel: device.createBuffer({ size: MAXN * 8, usage: S | GPUBufferUsage.COPY_SRC | GPUBufferUsage.COPY_DST }),
      pv: device.createBuffer({ size: MAXN * 8, usage: S | GPUBufferUsage.COPY_SRC | GPUBufferUsage.COPY_DST }),
      acc: device.createBuffer({ size: MAXN * 8, usage: S | GPUBufferUsage.COPY_SRC }),
      flags: device.createBuffer({ size: MAXN * 4, usage: S | GPUBufferUsage.COPY_SRC | GPUBufferUsage.COPY_DST }),
      cellCount: device.createBuffer({ size: CELLS * 4, usage: S }),
      cellStart: device.createBuffer({ size: CELLS * 4, usage: S }),
      cellCursor: device.createBuffer({ size: CELLS * 4, usage: S }),
      sorted: device.createBuffer({ size: MAXN * 4, usage: S }),
      events: device.createBuffer({ size: MAXN * 4, usage: S }),
      summary: device.createBuffer({ size: 256, usage: S | GPUBufferUsage.COPY_SRC }),
    };
    uni = device.createBuffer({ size: UNI_BYTES, usage: GPUBufferUsage.UNIFORM | GPUBufferUsage.COPY_DST });
    const off = stagingOffsets();
    for (let k = 0; k < 2; k++)
      staging[k] = device.createBuffer({ size: off.total, usage: GPUBufferUsage.MAP_READ | GPUBufferUsage.COPY_DST });
    const mk = (b) => ({ buffer: buf[b] });
    bindGroup = device.createBindGroup({ layout, entries: [
      { binding: 0, resource: { buffer: uni } },
      { binding: 1, resource: mk("pos") }, { binding: 2, resource: mk("vel") },
      { binding: 3, resource: mk("pv") }, { binding: 4, resource: mk("acc") },
      { binding: 5, resource: mk("flags") }, { binding: 6, resource: mk("cellCount") },
      { binding: 7, resource: mk("cellStart") }, { binding: 8, resource: mk("cellCursor") },
      { binding: 9, resource: mk("sorted") }, { binding: 10, resource: mk("events") },
      { binding: 11, resource: { buffer: buf.summary } },
    ]});
    uploadBirds();
    readyFlag = true;
    return true;
  }

  function uploadBirds() {                   // CPU 侧状态 -> GPU(初始化/重置/补充后调用)
    if (!device) return;
    const n2 = N * 2;
    device.queue.writeBuffer(buf.pos, 0, P.subarray(0, n2));
    device.queue.writeBuffer(buf.vel, 0, V.subarray(0, n2));
    device.queue.writeBuffer(buf.pv, 0, PV.subarray(0, n2));
    const fl = new Uint32Array(N);
    for (let i = 0; i < N; i++) fl[i] = (alive[i] ? 1 : 0) | (isMobber[i] ? 2 : 0) | (startled[i] ? 4 : 0);
    device.queue.writeBuffer(buf.flags, 0, fl);
  }

  function submit() {                        // 一帧物理: 8 个 dispatch + staging 拷贝, 一次提交
    if (!device) return;
    const u = new ArrayBuffer(UNI_BYTES);
    const U32 = new Uint32Array(u), F = new Float32Array(u);
    U32[0] = N; U32[1] = topoK; U32[2] = frame >>> 0;
    F[4] = migX; F[5] = migY;
    for (let t = 0; t < 3; t++) {
      const f = threats[t], o = 8 + t * 8;
      F[o] = f.x; F[o + 1] = f.y; F[o + 2] = f.vx; F[o + 3] = f.vy;
      F[o + 4] = f.dashT; F[o + 5] = f.lockIdx;
    }
    device.queue.writeBuffer(uni, 0, u);
    const enc = device.createCommandEncoder();
    const pass = enc.beginComputePass();
    pass.setBindGroup(0, bindGroup);
    const wg = Math.ceil(N / 256);
    pass.setPipeline(pipe.clearGrid);        pass.dispatchWorkgroups(1);
    pass.setPipeline(pipe.countCell);        pass.dispatchWorkgroups(wg);
    pass.setPipeline(pipe.scanCells);        pass.dispatchWorkgroups(1);
    pass.setPipeline(pipe.scatter);          pass.dispatchWorkgroups(wg);
    pass.setPipeline(pipe.neighborForces);   pass.dispatchWorkgroups(wg);
    pass.setPipeline(pipe.finalizeCentroid); pass.dispatchWorkgroups(1);
    pass.setPipeline(pipe.reduceSummary);    pass.dispatchWorkgroups(1);
    pass.setPipeline(pipe.integrate);        pass.dispatchWorkgroups(wg);
    pass.end();
    const off = stagingOffsets(), st = staging[cur];
    const n2b = N * 8, nb = N * 4;
    enc.copyBufferToBuffer(buf.pos, 0, st, off.pos, n2b);
    enc.copyBufferToBuffer(buf.vel, 0, st, off.vel, n2b);
    enc.copyBufferToBuffer(buf.pv, 0, st, off.pv, n2b);
    enc.copyBufferToBuffer(buf.flags, 0, st, off.flags, nb);
    enc.copyBufferToBuffer(buf.events, 0, st, off.events, nb);
    enc.copyBufferToBuffer(buf.summary, 0, st, 0, 256);
    device.queue.submit([enc.finish()]);
    submittedN = N; everSubmitted = true;
    cur = 1 - cur;
  }

  async function readbackPrev() {            // 取上一帧提交的回读(mapAsync 等待完成)
    if (!device || !everSubmitted) return null;
    const other = 1 - cur;
    const st = staging[other];
    await st.mapAsync(GPUMapMode.READ);
    const ab = st.getMappedRange();
    const off = stagingOffsets();
    const n = submittedN;
    P.set(new Float32Array(ab, off.pos, n * 2));
    V.set(new Float32Array(ab, off.vel, n * 2));
    PV.set(new Float32Array(ab, off.pv, n * 2));
    const FL = new Uint32Array(ab, off.flags, n);
    const cnt = new Uint32Array(ab, 0, 16);
    let sx = 0, sy = 0, c = 0;
    for (let i = 0; i < n; i++) {
      const f = FL[i];
      alive[i] = f & 1;
      isMobber[i] = (f >> 1) & 1;
      startled[i] = (f >> 2) & 1;
      if (f & 1) { sx += P[2*i]; sy += P[2*i+1]; c++; }
    }
    const meta = {
      n, cnt: cnt[0],
      cx: c ? sx / c : W / 2, cy: c ? sy / c : H / 2,
      events: Math.min(cnt[2], n),
      evIdx: new Uint32Array(ab, off.events, n).slice(0, Math.min(cnt[2], n)),
      kills: [cnt[6], cnt[7], cnt[8]],
      dashEats: [cnt[9], cnt[10], cnt[11]],
      mob: [cnt[12], cnt[13], cnt[14]],
      hotspot: cnt[15],
    };
    st.unmap();
    return meta;
  }

  return {
    init, submit, readbackPrev, uploadBirds,
    get ready() { return readyFlag; },
    info() { return device ? { cells: CELLS, stagingBytes: stagingOffsets().total } : null; },
  };
})();

 交换、最后一趟绑回默认帧缓冲做拷贝、SIM.perf 三个字段。

 ============================ 探索性观察: topoK 与密度 ============================
 理论预言"邻居数恒定 -> 群体形态应与 N 无关"。无猛禽干扰各跑 30 秒后:
        N      最近邻    第7近邻   极化度   展布    展布/第7近邻
      2000     11.4px    21.3px    0.893   215px      10.12
     20000      6.2px    12.0px    0.896   351px      29.18
 结论:**局部结构确实拓扑化**(最近邻与第7近邻随 N 变小, 但都与 1/√N 同量级, 邻居数恒定),
 但**全局形态与 N 相关** —— 归一化的"展布/第7近邻"从 10.1 涨到 29.2。原因是凝聚力/迁徙/
 软边界/分离半径(SEP_R=20px)都是**绝对像素尺度**, 只有"选邻居"这一步是拓扑的。所以严格
 意义上的"形态与 N 无关"在本实现里被证伪; 想让它成立需要把 SEP/COHESION 也换成相对尺度。

 ============================ 第 4 轮之一: 采样目标获取 ============================
 问题: 最密网格桶有量化误差; 质心可能正好落在隼自己犁出的空腔里; 也无法扩展风险项。
 做法(P1): perceive() 在收集可见椋鸟的同时, 以猛禽为原点布 5 环 × 16 角 = 80 个极坐标
   候选点, 每只鸟累加进"距离 < SCOOP_R 的候选点", 得到一个 80 维的"捞勺计数"。
   打分公式:
     score(c) = (c 的捞勺内鸟数) × (1 + DIR_LAG·cos(候选方向, 当前航向))
                × [锥内乘项]        // 见第 4 轮之二 D
                // 扩展位: 雁权重 / 威慑锥扣分 / mob 扣分
   方向滞后项让目标不会在两个等分候选点之间抖动; 锥内乘项让"选目标"自己绕开雁群正面,
   而不是写一条"猛禽看见雁就躲"的硬编码。
   取 argmax 候选点为冲刺中心 (cx,cy); LOCK_R=40px 内最近的鸟为 lockIdx,
   40px 内没有鸟就放弃这次冲刺(不做无效冲刺)。tryDash 的 lockIdx 语义不变:
   冲刺途中只吃得到被锁定的那一只(雁的 EAT 判定不加, 雁大不存在迷惑);
   互攻/驱逐冲刺传 -1, 冲刺结束 lockIdx 归 -1。
   常量: SAMPLE_RADII={80,160,240,320,400} SAMPLE_ANGLES=16 SCOOP_R=60 LOCK_R=40
         DIR_LAG=0.3  (候选距猛禽 > VIEW_R - SCOOP_R 的直接跳过)
   性能: 内层用"半径平方 + 预计算角度窗(CAND_W)"把每只鸟的候选测试从 80 次压到 5~10 次,
   N=2000 下整步约 9~11ms。

 ============================ 第 4 轮之二: 雁群中心化重构 ============================
 目标: 雁从"有槽位的椋鸟"变成真正的中心化组织 —— 响应即时、无单点故障、有威慑,
   但保留制度性弱点。
 A 领队智能: LEADER_SENSE_R=400 内最近猛禽 -> leaderAng 以 LEADER_MAX_TURN=0.04 rad/帧
   限速转向远离; 无威胁时随机游动 + 向心; 威胁 < LEADER_PANIC_R=250 时全队速度
   ×ESCAPE_HUSTLE=1.15。每只雁另有 GOOSE_MAX_TURN=0.06 rad/帧 的转向限速(压坡惯性)。
 B 轮值领队: ROTATE_FRAMES=1200(20 秒)换一次 apex; 槽位号 slot(i) = (i + gooseShift) % M,
   虚拟领队位置就是当前 apex 雁的位置(所以换岗不瞬移, leaderAng 继承)。没有可斩首的固定领队。
   幼雁内道靠"残类"实现: 幼雁是固定个体(i % 4 === 0, 25%), 而 4 | M=60, 所以它们的槽位
   残类恒等于 gooseShift % 4 —— 把这一残类的槽位排成内道(小排号), 幼雁就永远在内道。
 C 核心保护: JUVENILE_FRAC=0.25; 幼雁死亡 -> JUVENILE_DEBUFF_FRAMES=1800 帧内全体归位力
   ×DEBUFF_FACTOR=0.7(沮丧期)。
 D 威慑锥: 顶点 = V 尖, 轴 = 领队航向, 半角 60°, 半径 CONE_R=180。两个效果:
   a) 锥内候选点打分 ×CONE_SCORE_FACTOR=0.25(与之一咬合, 选目标自己绕开正面);
   b) 锥内猛禽持续失血: 隼冲刺能耗 ×CONE_FALCON_DASH_MULT=2、鹰每帧 pending -= 0.01
      (走 pending 才进得了 Q 表 —— 威慑是"被定价", 不是硬编码行为)。锥只罩正面 120°,
      绕后不受罚, 保留渗透缝隙。
 E 斩首混乱: 当前 apex 被吃的瞬间 disarray = DISARRAY_FRAMES=90: 槽位力关闭, 全体退化为
   个体径向逃逸(复用逃逸路径), 结束后按新 apex 重建。这是雁相对椋鸟群的结构性弱点。

 ============================ 第 4 轮之三: 隼的策略进化 ============================
 原则: 制度(能量轮换/拦截位/交棒/驱逐)不学, 学的是制度内的操作参数 —— 鹰学价值(Q-learning),
   隼学策略(进化), 两种学习哲学同框。
 可学习参数(每只隼一份, 归一化 [0,1] 存储):
   p1 出手能量阈值 -> DASH_MIN     ∈[20,60] 默认 35
   p2 交棒能量阈值 -> ENERGY_LOW   ∈[10,40] 默认 25
   p3 双打能量阈值 -> DOUBLE_TEAM_E∈[60,95] 默认 85
   p4 拦截位外推   -> INTERCEPT_AHEAD ∈[0.3,0.8] 默认 0.55
   p5 mob 容忍     -> 乘在 DASH_DRAIN 的 mob 项上 ∈[0,2] 默认 1.0
   p6 线性冲刺策略(4 维特征 + 阈值)留作扩展位, 未实现 —— 若要做, 在 6.8 的出手判定处
      把 PV_.dashMin 比较换成 w·[能量, 目标密度, 1-mob, 航向对齐度] > θ, 并把 w/θ 塞进 ES 向量。
 算法: (1+1)-ES 在线进化, ES_EVAL_FRAMES=3600(60 秒)一个评估期(适应度噪声大, 期长配对比较)。
   双缓冲: 偶数期评估当前参数, 奇数期评估"当前 + N(0, ES_SIGMA=0.1) 截断到 [0,1]"的候选;
   期末配对比较, 候选更好就采纳(p = cand, gen++), 否则回滚。两只隼各自抽噪声, 随机源不共享。
   适应度(本期净值):
     fit = Δscore - ES_MISS_WEIGHT(=2)·Σ落空罚款 - ES_METAB_WEIGHT(=5)·Σ代谢成本
   无价值函数、无状态抽象, 纯爬山 —— 允许学成"永不出手"的局部最优(此时 fit≈0, 不出手
   既不赚也不罚), 那是诚实的失败模式, 不是 bug。
   观测: 面板「重置隼策」按钮 / SIM.falconPolicies / 每只隼能量条上方的"策略 n 代"小字。

 ============================ 第 4 轮之四: 雁的借网预警 ============================
 雁从"只看 150px 内的猛禽本体", 扩展为接入椋鸟群(2000 节点分布式传感器网)。
 1 惊飞度: 新增 PV 数组存上帧速度; 判据 = 本帧与上帧速度夹角 > 60°(cos < PV_ALERT_COS=0.5)。
   【陷阱】必须在位置更新之后、覆盖 PV 之前算 —— 顺序错一帧(先覆盖再比)等于拿自己比自己,
   全体误报。另外不用速度大小当指标: 椋鸟平时也贴满速。
 2 雁判定: 用现有网格查周边 3×3 桶, ALERT_R=140 内惊飞椋鸟 >= ALERT_N=3 -> 视同猛禽进入
   FORMATION_BREAK: 编队解散, 逃逸方向 = 惊飞椋鸟的平均速度方向(它们逃向即远离猛禽方向,
   直接跟流)。与自身 150px 的猛禽感知取并集。
 3 涌现代价(不修): 椋鸟翻滚会致误报, 雁散一下又合拢 —— 这是偷听公共信息的真实成本。
   ALERT_N 调高误报少但预警慢。面板能看到"借网预警 n 次(误报 m)"。
